package semanticcmd

import (
	"context"
	"fmt"
	"strings"
	"time"

	"moedex/internal/semanticrun"
)

const DefaultCaptureTimeout = 5 * time.Minute

// CaptureOptions makes the compiler execution boundary explicit. The runner
// resolves managed acquisition identity or verifies an explicitly pinned checkout.
type CaptureOptions = semanticrun.Options

type CaptureSummary struct {
	Summary
	Artifact                  string `json:"artifact"`
	Workspace                 string `json:"workspace"`
	Repo                      string `json:"repo"`
	Commit                    string `json:"commit"`
	Origin                    string `json:"origin,omitempty"`
	ProjectID                 int64  `json:"project_id"`
	DependencyBundleSHA256    string `json:"dependency_bundle_sha256,omitempty"`
	RestoreStandardEvaluation bool   `json:"restore_standard_evaluation,omitempty"`
	DependencyByteLimit       int64  `json:"dependency_byte_limit"`
}

// Capture delegates projection, bounded process execution, validation and
// exclusive artifact publication to semanticrun. The returned workspace is the
// retained projection root used by later explicit attachment revalidation.
func Capture(ctx context.Context, opts CaptureOptions) (CaptureSummary, error) {
	if err := ctx.Err(); err != nil {
		return CaptureSummary{}, err
	}
	if err := validateCaptureOptions(opts); err != nil {
		return CaptureSummary{}, err
	}
	result, err := semanticrun.Capture(ctx, opts)
	if err != nil {
		return CaptureSummary{}, err
	}
	return CaptureSummary{
		Summary: summarize(result.Artifact), Artifact: result.Path,
		Workspace: result.Workspace, Repo: result.Repo,
		Commit: result.Commit, Origin: result.Origin, ProjectID: result.ProjectID,
		DependencyBundleSHA256:    result.DependencyBundleSHA256,
		RestoreStandardEvaluation: result.RestoreStandardEvaluation,
		DependencyByteLimit:       result.DependencyByteLimit,
	}, nil
}

func validateCaptureOptions(opts CaptureOptions) error {
	if !opts.RestoreOffline {
		return fmt.Errorf("semantic capture: --restore-offline must be explicitly enabled")
	}
	if opts.Timeout <= 0 {
		return fmt.Errorf("semantic capture: timeout must be positive")
	}
	managed, checkout := strings.TrimSpace(opts.ManagedRoot) != "", strings.TrimSpace(opts.Checkout) != ""
	if managed == checkout {
		return fmt.Errorf("semantic capture: exactly one of --managed-root or --checkout is required")
	}
	if managed && (opts.Commit != "" || opts.Origin != "") {
		return fmt.Errorf("semantic capture: --commit and --origin require --checkout")
	}
	if checkout && (strings.TrimSpace(opts.Commit) == "" || strings.TrimSpace(opts.Origin) == "") {
		return fmt.Errorf("semantic capture: --checkout requires --commit and --origin")
	}
	if opts.MaxProjectionBytes < 0 || opts.MaxProjectionBytes > 1<<30 {
		return fmt.Errorf("semantic capture: --max-projection-bytes must be between 0 and 1073741824")
	}
	for _, field := range []struct{ name, value string }{
		{"repo", opts.Repo},
		{"project", opts.Project}, {"framework", opts.Framework},
		{"dotnet", opts.Dotnet}, {"worker", opts.Worker},
		{"sdk-path", opts.SDKPath}, {"workspace", opts.Workspace},
		{"output", opts.Output},
	} {
		if strings.TrimSpace(field.value) == "" {
			return fmt.Errorf("semantic capture: --%s is required", field.name)
		}
	}
	return nil
}
