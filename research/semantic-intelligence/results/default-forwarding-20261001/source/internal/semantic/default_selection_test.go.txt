package semantic

import (
	"encoding/json"
	"testing"
)

func TestDefaultSelectionValidation(t *testing.T) {
	symbol := func(desc string) Symbol {
		s := Symbol{Key: SymbolKey{Language: "csharp", NamespaceKind: "project", Namespace: "repo/app", DescriptorKind: "documentation_comment_id", Descriptor: desc}}
		s.ID = s.ComputeID()
		return s
	}
	typ, iface, template, arg := symbol("T:Handler"), symbol("M:Contracts.IBase.Run(System.Object)"), symbol("M:Contracts.IHandler`1.Contracts#IBase#Run(System.Object)"), symbol("T:Message")
	raw, _ := json.Marshal(map[string]any{"definition": template.Key.Descriptor, "arguments": []SymbolKey{arg.Key}})
	closed := template
	closed.Key.DescriptorKind = "constructed_interface_method_v1"
	closed.Key.Descriptor = string(raw)
	closed.ID = closed.ComputeID()
	fact := ImplementationFact{Kind: "interface_default_selection", Rule: "csharp-interface-selection-v1", EvidenceScope: "compile_time", InterfaceSymbolID: iface.ID, ImplementingTypeSymbolID: typ.ID, SelectedDefaultSymbolID: closed.ID, DefaultTemplateSymbolID: template.ID}
	if err := ValidateImplementationFacts([]ImplementationFact{fact}); err != nil {
		t.Fatal(err)
	}
	if err := ValidateImplementationSymbols(fact, typ, iface, typ, closed, template); err != nil {
		t.Fatal(err)
	}
	for _, v := range []string{"6", "7", "8", "9"} {
		if (ValidateImplementationExtractor(fact, v) == nil) != (v == "8" || v == "9") {
			t.Fatal("extractor gate", v)
		}
	}
	for name, mutate := range map[string]func(*ImplementationFact){
		"missing closed":   func(f *ImplementationFact) { f.SelectedDefaultSymbolID = "" },
		"missing template": func(f *ImplementationFact) { f.DefaultTemplateSymbolID = "" },
		"wrong kind":       func(f *ImplementationFact) { f.Kind = "interface_method_implementation" },
		"old rule":         func(f *ImplementationFact) { f.Rule = "csharp-interface-default-v1" },
		"runtime claim":    func(f *ImplementationFact) { f.EvidenceScope = "runtime" },
	} {
		t.Run(name, func(t *testing.T) {
			f := fact
			mutate(&f)
			if ValidateImplementationFacts([]ImplementationFact{f}) == nil {
				t.Fatal("invalid fact admitted")
			}
		})
	}
	for name, mutate := range map[string]func(*Symbol, *Symbol, *Symbol, *Symbol){
		"method anchor": func(a, i, c, s *Symbol) { *a = symbol("M:Handler.Run") },
		"wrong slot":    func(a, i, c, s *Symbol) { i.Key.Descriptor = "M:Other.IBase.Run(System.Object)" },
		"wrong body":    func(a, i, c, s *Symbol) { s.Key.Descriptor = "M:Other`1.Contracts#IBase#Run(System.Object)" },
		"wrong project": func(a, i, c, s *Symbol) { s.Key.Namespace = "repo/other" },
		"metadata body": func(a, i, c, s *Symbol) { c.Key.NamespaceKind = "assembly" },
		"open body":     func(a, i, c, s *Symbol) { *c = template },
	} {
		t.Run(name, func(t *testing.T) {
			a, i, c, s := typ, iface, closed, template
			mutate(&a, &i, &c, &s)
			if ValidateImplementationSymbols(fact, a, i, typ, c, s) == nil {
				t.Fatal("invalid symbols admitted")
			}
		})
	}
	if ValidateImplementationSymbols(fact, typ, iface, typ) == nil {
		t.Fatal("missing selection symbols admitted")
	}
}
