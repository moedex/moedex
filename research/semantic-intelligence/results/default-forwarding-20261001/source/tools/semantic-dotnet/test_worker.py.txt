#!/usr/bin/env python3
"""Offline worker integration. Builds/runs only disposable copied projects.

Usage: python3 tools/semantic-dotnet/test_worker.py --dotnet /path/dotnet --sdk-path /path/sdk/10.0.100
MSBuildWorkspace requires local build-host IPC; this test needs that permission.
"""
import argparse
import base64
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import tempfile

p = argparse.ArgumentParser()
p.add_argument('--dotnet', required=True)
p.add_argument('--sdk-path', required=True)
p.add_argument('--output-dir')
a = p.parse_args()
home = Path(a.output_dir or tempfile.mkdtemp(prefix='moedex-worker-test-')).resolve()
home.mkdir(parents=True, exist_ok=True)
repo = Path(__file__).resolve().parents[2]
shutil.copytree(Path(__file__).parent, home/'worker', dirs_exist_ok=True, ignore=shutil.ignore_patterns('bin', 'obj', '__pycache__'))
root = home/'fixture'
shutil.copytree(repo/'internal/eval/testdata/semantic-intelligence/compiler-v1', root, dirs_exist_ok=True)
shutil.copy(Path(__file__).parent/'NuGet.Config', root/'NuGet.Config')
(root/'global.json').write_text(json.dumps({'sdk': {'version': Path(a.sdk_path).name, 'rollForward': 'disable'}}))
env = dict(os.environ, DOTNET_ROOT=str(Path(a.dotnet).parent), DOTNET_CLI_HOME=str(home/'cli'), NUGET_PACKAGES=str(home/'packages'), DOTNET_NOLOGO='1', DOTNET_CLI_TELEMETRY_OPTOUT='1', MSBUILDDISABLENODEREUSE='1')

def run(args, expected=0):
    result = subprocess.run([a.dotnet, *map(str, args)], env=env, capture_output=True, timeout=120)
    if result.returncode != expected:
        raise AssertionError((args, result.returncode, result.stdout.decode(errors='replace')[-4000:], result.stderr.decode(errors='replace')))
    return result.stdout

run(['build', home/'worker/Moedex.SemanticWorker.csproj', '--nologo', '-p:UseSharedCompilation=false'])
dll = home/'worker/bin/Debug/net10.0/Moedex.SemanticWorker.dll'

def extract(project, expected=0, config='Debug'):
    run(['restore', root/project, '--nologo'])
    output = run([dll, '--repo', 'fixture', '--root', root, '--project', project, '--framework', 'net10.0', '--sdk-path', a.sdk_path, '--configuration', config], expected)
    rows = [json.loads(line) for line in output.splitlines()]
    assert rows[-1]['record_type'] == 'stream_summary'
    assert all(r['schema'] == 'moedex.semantic-worker.v1' for r in rows)
    projects = {r['project']: r for r in rows if r['record_type'] == 'project'}
    assert rows[-1]['projects'] == len(projects)
    source_bytes = {}
    for key, r in projects.items():
        assert hashlib.sha256(r['capture_json'].encode()).hexdigest() == r['build_context']
        capture = json.loads(r['capture_json'])
        assert capture['project'] == key
        assert [(s['path'], s['sha256'], s['byte_size']) for s in capture['sources']] == [(s['path'], s['sha256'], s['byte_size']) for s in r['sources']]
        for s in r['sources']:
            b = base64.b64decode(s['content_base64']) if s['content_base64'] is not None else (root/s['path']).read_bytes()
            assert len(b) == s['byte_size'] and hashlib.sha256(b).hexdigest() == s['sha256']
            source_bytes[key, s['path']] = b
    counts = {'declaration': 0, 'reference': 0}
    for row in rows:
        if row['record_type'] not in counts:
            continue
        counts[row['record_type']] += 1
        assert row['build_context'] == projects[row['project']]['build_context']
        b = source_bytes[row['project'], row['source_path']]
        start, length = row['span']['byte_offset'], row['span']['byte_length']
        assert b[start:start+length].decode() == row['source_text']
        assert hashlib.sha256(b).hexdigest() == row['source_sha256']
        for symbol in [row['symbol'], *row['candidates']]:
            if symbol:
                assert symbol['descriptor_kind'] in ('documentation_comment_id', 'location_fallback', 'constructed_interface_method_v1')
        for fact in row.get('domain_facts', []):
            assert row['record_type'] == 'reference' and row['binding_status'] == 'resolved'
            assert projects[row['project']]['compilation_status'] == 'complete'
            assert fact['rule'] == 'csharp-framework-v1' and fact['evidence_scope'] == 'compile_time'
            roles = {'di_registration': ['service', 'implementation'],
                     'message_publish': ['message'], 'message_consumer': ['message'],
                     'storage_entity': ['entity'], 'storage_table': ['entity']}
            assert [target['role'] for target in fact['targets']] == roles[fact['kind']]
            assert all(target['symbol']['descriptor_kind'] == 'documentation_comment_id' for target in fact['targets'])
    assert counts == {'declaration': rows[-1]['declarations'], 'reference': rows[-1]['references']}
    for summary in [r for r in rows if r['record_type'] == 'summary']:
        for kind, plural in [('declaration', 'declarations'), ('reference', 'references')]:
            assert summary[plural] == sum(r['record_type'] == kind and r.get('project') == summary['project'] for r in rows)
    return output, rows

all_rows = []
for project in ['App/App.csproj', 'AppFast/AppFast.csproj', 'ContextA/ContextA.csproj', 'ContextB/ContextB.csproj', 'Broken/Broken.csproj']:
    output, rows = extract(project, 2 if project.startswith('Broken') else 0)
    (home/(Path(project).parent.name+'.jsonl')).write_bytes(output)
    all_rows += rows
gold = json.loads((root/'gold.json').read_text())
project_paths = {x['id']: x['project_file'] for x in gold['projects']}

def anchored(kind, project, anchor):
    data = (root/anchor['file']).read_bytes()
    lines = data.splitlines(keepends=True)
    offset = sum(map(len, lines[:anchor['line']-1])) + lines[anchor['line']-1].index(anchor['text'].encode())
    return [r for r in all_rows if r['record_type'] == kind and r['project'] == project_paths[project] and r['source_path'] == anchor['file'] and r['span']['byte_offset'] == offset and r['source_text'] == anchor['text']]

def key(symbol):
    return (symbol['namespace_kind'], symbol['namespace'], symbol['descriptor'])

for label in gold['references']:
    matches = anchored('reference', label['project'], label['source'])
    if label['status'] == 'inactive':
        assert not matches, label['id']
        continue
    assert matches, label['id']
    if label['status'] == 'resolved':
        target = label['target']
        expected = {key(r['symbol']) for r in anchored('declaration', target['project'], target['anchor'])}
        assert any(r['binding_status'] == 'resolved' and key(r['symbol']) in expected for r in matches), label['id']
    elif label['status'] == 'external':
        assert any(r['symbol'] and r['symbol']['namespace_kind'] == 'assembly' and r['symbol']['descriptor'] == label['external_symbol'] for r in matches), label['id']
    elif label['status'] == 'ambiguous':
        expected = {key(r['symbol']) for target in label['candidate_targets'] for r in anchored('declaration', target['project'], target['anchor'])}
        assert any(r['binding_status'] == 'ambiguous' and {key(s) for s in r['candidates']} == expected for r in matches), label['id']
    else:
        assert any(r['binding_status'] == 'unresolved' and r['symbol'] is None for r in matches), label['id']
first, rows = extract('App/App.csproj')
second, _ = extract('App/App.csproj')
assert first == second, 'identical restored inputs must produce byte-identical output'
debug = {r['project']: r['build_context'] for r in rows if r['record_type'] == 'project'}
_, release = extract('App/App.csproj', config='Release')
assert debug['App/App.csproj'] != next(r['build_context'] for r in release if r['record_type']=='project' and r['project']=='App/App.csproj')
assert {r['code'] for r in all_rows if r['record_type']=='diagnostic' and r['project']=='Broken/Broken.csproj'} >= {'CS0103', 'CS0121'}

# Aggregate the two linked-file contexts in one actual workspace graph.
(root/'Both').mkdir(exist_ok=True)
(root/'Both/Both.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup><ItemGroup><ProjectReference Include="../ContextA/ContextA.csproj"/><ProjectReference Include="../ContextB/ContextB.csproj"/></ItemGroup></Project>')
output, rows = extract('Both/Both.csproj')
(home/'Both.jsonl').write_bytes(output)
symbols = [r['symbol']['namespace'] for r in rows if r['record_type']=='reference' and r['source_path']=='Shared/LinkedCall.cs' and r['source_text']=='Resolve']
assert len(set(symbols)) == 2

# BOM, multibyte scalars, CRLF and missing assets are independent controls.
(root/'Bytes').mkdir(exist_ok=True)
(root/'Bytes/Bytes.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup></Project>')
(root/'Bytes/B.cs').write_bytes(b'\xef\xbb\xbf' + 'class B { static void M() {} static void N() { /* \u03c0\U0001f600 */ M(); } }\r\n'.encode())
extract('Bytes/Bytes.csproj')
assets = root/'Bytes/obj/project.assets.json'
assets.unlink()
missing = run([dll, '--repo', 'fixture', '--root', root, '--project', 'Bytes/Bytes.csproj', '--framework', 'net10.0', '--sdk-path', a.sdk_path], 2)
assert json.loads(missing.splitlines()[-1])['compilation_status'] == 'incomplete'

# Physical inputs outside the projection (direct or through symlinks) must not
# masquerade as generated bytes. A failed stream never carries a terminal row.
(home/'Outside.cs').write_text('class Outside {}')
(root/'Escape').mkdir(exist_ok=True)
escape = root/'Escape/Escape.csproj'
for include in ['../../Outside.cs', 'Link.cs']:
    escape.write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework><EnableDefaultCompileItems>false</EnableDefaultCompileItems></PropertyGroup><ItemGroup><Compile Include="'+include+'"/></ItemGroup></Project>')
    if include == 'Link.cs':
        link = root/'Escape/Link.cs'
        if not link.exists(): link.symlink_to(home/'Outside.cs')
    run(['restore', escape, '--nologo'])
    failed = run([dll, '--repo', 'group/fixture', '--root', root, '--project', 'Escape/Escape.csproj', '--framework', 'net10.0', '--sdk-path', a.sdk_path], 1)
    assert b'"record_type":"stream_summary"' not in failed

# Reference aliases are evaluated semantic input, retained in capture.
(root/'Alias').mkdir(exist_ok=True)
(root/'Alias/Alias.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup><ItemGroup><ProjectReference Include="../Contracts/Contracts.csproj" Aliases="custom"/></ItemGroup></Project>')
_, aliases = extract('Alias/Alias.csproj')
capture = json.loads(next(r['capture_json'] for r in aliases if r['record_type']=='project' and r['project']=='Alias/Alias.csproj'))
assert capture['project_references'][0]['aliases'] == ['custom']
run(['restore', root/'Bytes/Bytes.csproj', '--nologo'])
namespaced = run([dll, '--repo', 'group/fixture', '--root', root, '--project', 'Bytes/Bytes.csproj', '--framework', 'net10.0', '--sdk-path', a.sdk_path])
assert any(r['symbol']['namespace'] == 'group/fixture/Bytes/Bytes.csproj' for r in map(json.loads, namespaced.splitlines()) if r['record_type']=='declaration')

# Primary constructor declarations overlap their type name but must retain a
# distinct occurrence identity, including records and partial declarations.
(root/'Primary').mkdir(exist_ok=True)
(root/'Primary/Primary.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup></Project>')
(root/'Primary/P.cs').write_text('''class Outer {
    public partial class Nested(int value) { public int Value => value; }
    public partial class Nested { public Nested() : this(0) {} }
    public record R(int Value);
    public struct S(int value) { public int Value => value; }
    object Make() => new Nested(1);
    object Record() => new R(2);
    object Struct() => new S(3);
}''')
output, primary = extract('Primary/Primary.csproj')
(home/'Primary.jsonl').write_bytes(output)
for descriptor in ['M:Outer.Nested.#ctor(System.Int32)', 'M:Outer.R.#ctor(System.Int32)', 'M:Outer.S.#ctor(System.Int32)']:
    declarations = [r for r in primary if r['record_type']=='declaration' and r['symbol']['descriptor']==descriptor]
    references = [r for r in primary if r['record_type']=='reference' and (r.get('symbol') or {}).get('descriptor')==descriptor and r['reference_kind']=='constructor']
    assert len(declarations)==1 and len(references)==1, (descriptor, declarations, references)
    declaration = declarations[0]
    assert declaration['reference_kind']=='constructor_declaration'
    assert any(r['record_type']=='declaration' and r['span']==declaration['span'] and r['source_path']==declaration['source_path'] and r['reference_kind']=='declaration' and r['symbol']['kind']=='NamedType' for r in primary)
assert len([r for r in primary if r['record_type']=='declaration' and r['symbol']['descriptor']=='M:Outer.Nested.#ctor'])==1

# A real SDK source generator, rather than a supplied .g.cs filename.
(root/'Generated').mkdir(exist_ok=True)
(root/'Generated/Generated.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup></Project>')
(root/'Generated/R.cs').write_text('using System.Text.RegularExpressions; public partial class R { [GeneratedRegex("a")] public static partial Regex Make(); }')
output, generated = extract('Generated/Generated.csproj')
(home/'Generated.jsonl').write_bytes(output)
project = next(r for r in generated if r['record_type']=='project')
virtual = [s for s in project['sources'] if s['path'].startswith('.generated/')]
assert virtual and all(s['generated'] and s['content_base64'] for s in virtual)
assert any(r['record_type']=='declaration' and r['source_path'].startswith('.generated/') for r in generated)

# Generator exceptions do not appear in Compilation.GetDiagnostics by themselves.
# The verification driver must surface this failure even if nothing needs output.
(root/'ThrowGenerator').mkdir(exist_ok=True)
compiler_dir = Path(a.sdk_path)/'Roslyn/bincore'
(root/'ThrowGenerator/ThrowGenerator.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup><ItemGroup><Reference Include="'+str(compiler_dir/'Microsoft.CodeAnalysis.dll')+'"/><Reference Include="'+str(compiler_dir/'Microsoft.CodeAnalysis.CSharp.dll')+'"/></ItemGroup></Project>')
(root/'ThrowGenerator/Generator.cs').write_text('using Microsoft.CodeAnalysis; [Generator] public class Boom : ISourceGenerator { public void Initialize(GeneratorInitializationContext c) {} public void Execute(GeneratorExecutionContext c) { throw new System.Exception("fixture generator failure"); } }')
run(['build', root/'ThrowGenerator/ThrowGenerator.csproj', '--nologo', '-p:UseSharedCompilation=false'])
(root/'ThrowApp').mkdir(exist_ok=True)
(root/'ThrowApp/ThrowApp.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net10.0</TargetFramework></PropertyGroup><ItemGroup><Analyzer Include="../ThrowGenerator/bin/Debug/net10.0/ThrowGenerator.dll"/></ItemGroup></Project>')
(root/'ThrowApp/A.cs').write_text('class A {}')
output, failed_generator = extract('ThrowApp/ThrowApp.csproj', 2)
(home/'ThrowApp.jsonl').write_bytes(output)
assert any(r['record_type']=='diagnostic' and r['code']=='generator_failure' for r in failed_generator)
assert any(r['record_type']=='diagnostic' and r['code']=='CS8785' for r in failed_generator)

# Stop draining stdout after the project capture. Remaining occurrence output is
# larger than the pipe buffer, holding the worker before its terminal recheck.
command = [a.dotnet, str(dll), '--repo', 'fixture', '--root', str(root), '--project', 'App/App.csproj', '--framework', 'net10.0', '--sdk-path', a.sdk_path]
process = subprocess.Popen(command, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
first = process.stdout.readline()
assert json.loads(first)['record_type'] == 'project'
assert process.poll() is None, 'mutation control must occur before worker completion'
source = root/'App/Calls.cs'
original = source.read_bytes()
try:
    source.write_bytes(original+b'\n// mutated after capture\n')
    tail, error = process.communicate(timeout=120)
    assert process.returncode == 1 and b'captured input changed' in error
    assert b'"record_type":"stream_summary"' not in first+tail
finally:
    source.write_bytes(original)
    if process.poll() is None:
        process.kill()
        process.communicate()
print('PASS: 19 fixture labels, context separation, raw UTF8 spans, capture integrity, deterministic output, configuration invalidation, incomplete restore, outside-source rejection, reference aliases, slash repo namespace, primary constructor declarations, SDK source generator, generator failure, concurrent input mutation; artifacts:', home)
