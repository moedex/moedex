func digest(b []byte) string { h := sha256.Sum256(b); return hex.EncodeToString(h[:]) }
func validPath(p string) bool {
	return p != "" && p != "." && path.Clean(p) == p && !strings.HasPrefix(p, "/") && p != ".." && !strings.HasPrefix(p, "../") && !strings.ContainsAny(p, "\\:\x00")
}
func maxBytes(o Options) int64 {
	if o.MaxBytes > 0 {
		return o.MaxBytes
	}
	return DefaultMaxBytes
}

// Import accepts complete or explicitly incomplete compiler captures. Callers
// must check Complete before advertising an artifact as a successful generation.
// Raw source content is verified through a confined filesystem root. Virtual
// generated inputs require producer-supplied bytes and remain marked generated.
func Import(ctx context.Context, input io.Reader, opts Options) (*semantic.Artifact, error) {
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	if opts.MaxBytes < 0 {
		return nil, fmt.Errorf("semantic import: invalid byte limit")
	}
	if !validPath(opts.Repo) || opts.Root == "" || opts.ProjectID < 0 {
		return nil, fmt.Errorf("semantic import: repo, root and nonnegative project identity required")
	}
	limit := maxBytes(opts)
	if limit <= 0 || limit > DefaultMaxBytes {
		return nil, fmt.Errorf("semantic import: invalid byte limit")
	}
	root, err := os.OpenRoot(opts.Root)
	if err != nil {
		return nil, err
	}
	defer root.Close()
	lr := &io.LimitedReader{R: input, N: limit + 1}
	scan := bufio.NewScanner(lr)
	scan.Buffer(make([]byte, 64<<10), int(limit+1))
	projects := map[string]record{}
	summaries := map[string]record{}
	counts := map[string][3]int{}
	var occurrences []record
	var diagnostics []record
	var terminal *record
	line := 0
	verifiedInputs := map[string]string{}
	var inputBytes int64
	for scan.Scan() {
		line++
		if err := ctx.Err(); err != nil {
			return nil, err
		}
		if terminal != nil {
			return nil, fmt.Errorf("semantic import: data after terminal record")
		}
		var r record
		if err := json.Unmarshal(scan.Bytes(), &r); err != nil {
			return nil, fmt.Errorf("semantic import line %d: %w", line, err)
		}
		if r.Schema != WorkerSchema {
			return nil, fmt.Errorf("semantic import: unsupported worker schema %q", r.Schema)
		}
		if len(r.ImplementationFacts) > 0 && r.Type != "declaration" {
			return nil, fmt.Errorf("semantic import: implementation facts require a declaration record")
		}
		if len(r.DomainFacts) > 0 && r.Type != "reference" {
			return nil, fmt.Errorf("semantic import: domain facts require a reference record")
		}
		switch r.Type {
		case "project":
			if !validPath(r.Project) || r.Repo != opts.Repo || r.Context != digest([]byte(r.CaptureJSON)) || !json.Valid([]byte(r.CaptureJSON)) {
				return nil, fmt.Errorf("semantic import: invalid project capture %q", r.Project)
			}
			if _, ok := projects[r.Project]; ok {
				return nil, fmt.Errorf("semantic import: duplicate project %q", r.Project)
			}
			if r.Extractor == "" || r.ExtractorVersion == "" || r.CompilerVersion == "" {
				return nil, fmt.Errorf("semantic import: missing extractor/compiler identity")
			}
			if err := verifyCapture(r); err != nil {
				return nil, err
			}
			if err := verifyInputs(ctx, root, r.CaptureJSON, limit, verifiedInputs, &inputBytes); err != nil {
				return nil, err
			}
			projects[r.Project] = r
		case "declaration", "reference":
			p, ok := projects[r.Project]
			if !ok || p.Context != r.Context || !consistent(p, r) {
				return nil, fmt.Errorf("semantic import: occurrence without matching project")
			}
			c := counts[r.Project]
			if r.Type == "declaration" {
				c[0]++
			} else {
				c[1]++
			}
			counts[r.Project] = c
			occurrences = append(occurrences, r)
		case "diagnostic":
			diagnostics = append(diagnostics, r)
			if p, ok := projects[r.Project]; !ok || p.Context != r.Context || !consistent(p, r) {
				return nil, fmt.Errorf("semantic import: diagnostic without matching project context")
			}
			if r.Code == "" || r.Severity == "" {
				return nil, fmt.Errorf("semantic import: invalid diagnostic")
			}
			if r.Severity == "error" {
				c := counts[r.Project]
				c[2]++
				counts[r.Project] = c
			}
		case "summary":
			p, ok := projects[r.Project]
			if !ok || p.Context != r.Context || !consistent(p, r) {
				return nil, fmt.Errorf("semantic import: summary without matching project")
			}
			if _, ok := summaries[r.Project]; ok {
				return nil, fmt.Errorf("semantic import: duplicate summary")
			}
			summaries[r.Project] = r
		case "stream_summary":
			copy := r
			terminal = &copy
		default:
			return nil, fmt.Errorf("semantic import: unsupported record type %q", r.Type)
		}
	}
	if err := scan.Err(); err != nil {
		return nil, fmt.Errorf("semantic import: read stream: %w", err)
	}
	if lr.N == 0 {
		return nil, fmt.Errorf("semantic import: stream exceeds byte limit")
	}
	if terminal == nil || len(projects) == 0 || terminal.Projects != len(projects) || len(summaries) != len(projects) {
		return nil, fmt.Errorf("semantic import: missing or inconsistent completion records")
	}
	declarations, references := 0, 0
	complete := true
	for name, p := range projects {
		s := summaries[name]
		c := counts[name]
		if p.Status != s.Status || s.Declarations != c[0] || s.References != c[1] || s.Errors != c[2] {
			return nil, fmt.Errorf("semantic import: inconsistent summary for %s", name)
		}
		if p.Status != "complete" {
			complete = false
		} else if s.Errors != 0 {
			return nil, fmt.Errorf("semantic import: complete project has errors")
		}
		declarations += c[0]
		references += c[1]
	}
	expectedStatus := "incomplete"
	if complete {
		expectedStatus = "complete"
	}
	if terminal.Status != expectedStatus || terminal.Declarations != declarations || terminal.References != references {
		return nil, fmt.Errorf("semantic import: inconsistent terminal summary")
	}
	return assemble(ctx, root, opts, projects, occurrences, diagnostics, limit)
}

// Redundant optional provenance must never contradict its owning project.
func consistent(project, row record) bool {
	return (row.Repo == "" || row.Repo == project.Repo) &&
		(row.Extractor == "" || row.Extractor == project.Extractor) &&
		(row.ExtractorVersion == "" || row.ExtractorVersion == project.ExtractorVersion) &&
		(row.CompilerVersion == "" || row.CompilerVersion == project.CompilerVersion) &&
		(row.Status == "" || row.Status == project.Status)
}

func verifyCapture(r record) error {
	var capture struct {
		Project string         `json:"project"`
		Sources []sourceRecord `json:"sources"`
	}
	if err := json.Unmarshal([]byte(r.CaptureJSON), &capture); err != nil {
		return err
	}
	if capture.Project != r.Project || len(capture.Sources) != len(r.Sources) {
		return fmt.Errorf("semantic import: capture source roster mismatch")
	}
	entries := map[string]sourceRecord{}
	for _, s := range capture.Sources {
		if _, ok := entries[s.Path]; ok {
			return fmt.Errorf("semantic import: duplicate captured source")
		}
		entries[s.Path] = s
	}
	seen := map[string]bool{}
	for _, s := range r.Sources {
		captured, ok := entries[s.Path]
		if !ok || captured.SHA != s.SHA || captured.ByteSize != s.ByteSize || captured.Generated != s.Generated || seen[s.Path] {
			return fmt.Errorf("semantic import: capture source mismatch: %s", s.Path)
		}
		seen[s.Path] = true
	}
	return nil
}

func readSource(root *os.Root, s sourceRecord, remaining int64) ([]byte, error) {
	if !validPath(s.Path) || s.ByteSize > uint64(remaining) {
		return nil, fmt.Errorf("semantic import: invalid source path or byte limit: %s", s.Path)
	}
	var raw []byte
	if strings.HasPrefix(s.Path, ".generated/") {
		if !s.Generated || s.ContentBase64 == nil {
			return nil, fmt.Errorf("semantic import: generated source bytes missing")
		}
		var err error
		raw, err = base64.StdEncoding.DecodeString(*s.ContentBase64)
		if err != nil {
			return nil, err
		}
	} else {
		f, err := root.Open(s.Path)
		if err != nil {
			return nil, fmt.Errorf("semantic import: source %s: %w", s.Path, err)
		}
		defer f.Close()
		st, err := f.Stat()
		if err != nil {
			return nil, err
		}
		if !st.Mode().IsRegular() || st.Size() != int64(s.ByteSize) {
			return nil, fmt.Errorf("semantic import: source size/type mismatch: %s", s.Path)
		}
		raw, err = io.ReadAll(io.LimitReader(f, remaining+1))
		if err != nil {
			return nil, err
		}
		if s.ContentBase64 != nil {
			embedded, err := base64.StdEncoding.DecodeString(*s.ContentBase64)
			if err != nil || !bytes.Equal(raw, embedded) {
				return nil, fmt.Errorf("semantic import: generated disk source differs from capture")
			}
		}
	}
	if uint64(len(raw)) != s.ByteSize || digest(raw) != s.SHA || !utf8.Valid(raw) {
		return nil, fmt.Errorf("semantic import: source hash/encoding mismatch: %s", s.Path)
	}
	return raw, nil
}

func assemble(ctx context.Context, root *os.Root, opts Options, projects map[string]record, rows []record, diagnosticRows []record, limit int64) (*semantic.Artifact, error) {
	names := make([]string, 0, len(projects))
	for name := range projects {
		names = append(names, name)
	}
	sort.Strings(names)
	rawSources := map[string][]byte{}
	sourceRecords := map[string]sourceRecord{}
	var total int64
	for _, name := range names {
		for _, s := range projects[name].Sources {
			if err := ctx.Err(); err != nil {
				return nil, err
			}
			if existing, ok := sourceRecords[s.Path]; ok {
				if existing.SHA != s.SHA || existing.ByteSize != s.ByteSize || existing.Generated != s.Generated {
					return nil, fmt.Errorf("semantic import: conflicting source occurrences")
				}
				if s.ContentBase64 != nil {
					embedded, err := base64.StdEncoding.DecodeString(*s.ContentBase64)
					if err != nil || !bytes.Equal(embedded, rawSources[s.Path]) {
						return nil, fmt.Errorf("semantic import: inconsistent repeated source bytes")
					}
				} else if strings.HasPrefix(s.Path, ".generated/") {
					return nil, fmt.Errorf("semantic import: repeated generated source bytes missing")
				}
				continue
			}
			raw, err := readSource(root, s, limit-total)
			if err != nil {
				return nil, err
			}
			total += int64(len(raw))
			rawSources[s.Path] = raw
			sourceRecords[s.Path] = s
		}
	}
	paths := make([]string, 0, len(sourceRecords))
	for name := range sourceRecords {
		paths = append(paths, name)
	}
	sort.Strings(paths)
	type sourceDigest struct {
		Path, SHA string
		Size      uint64
	}
	roster := make([]sourceDigest, 0, len(paths))
	for _, name := range paths {
		s := sourceRecords[name]
		roster = append(roster, sourceDigest{name, s.SHA, s.ByteSize})
	}
	encoded, _ := json.Marshal(roster)
	snapshot := semantic.SourceSnapshot{Repo: opts.Repo, Commit: strings.ToLower(opts.Commit), Origin: opts.Origin, InputFingerprint: digest(encoded)}
	if opts.ProjectID != 0 {
		snapshot.ProjectID = strconv.FormatInt(opts.ProjectID, 10)
	}
	snapshot.ID = snapshot.ComputeID()
	artifact := &semantic.Artifact{Version: semantic.FormatVersion, Snapshots: []semantic.SourceSnapshot{snapshot}}
	sources := map[string]semantic.Source{}
	for _, name := range paths {
		s := sourceRecords[name]
		source := semantic.Source{SnapshotID: snapshot.ID, Path: name, RawSHA256: s.SHA, ByteSize: s.ByteSize, Generated: s.Generated}
		if source.Generated {
			source.Content = rawSources[name]
		}
		source.ID = source.ComputeID()
		sources[name] = source
		artifact.Sources = append(artifact.Sources, source)
	}
	contexts := map[string]semantic.BuildContext{}
	visiting := map[string]bool{}
	var makeContext func(string) (semantic.BuildContext, error)
	makeContext = func(name string) (semantic.BuildContext, error) {
		if c, ok := contexts[name]; ok {
			return c, nil
		}
		if visiting[name] {
			return semantic.BuildContext{}, fmt.Errorf("semantic import: cyclic context dependencies")
		}
		visiting[name] = true
		p, ok := projects[name]
		if !ok {
			return semantic.BuildContext{}, fmt.Errorf("semantic import: absent dependency project %s", name)
		}
		c := semantic.BuildContext{SnapshotID: snapshot.ID, Project: name, InputFingerprint: p.Context, Extractor: p.Extractor, ExtractorVersion: p.ExtractorVersion, Status: p.Status, Issues: append([]string(nil), p.Issues...), Capture: p.CaptureJSON, DependencyBundleSHA256: opts.DependencyBundleSHA256}
		for _, s := range p.Sources {
			c.SourceIDs = append(c.SourceIDs, sources[s.Path].ID)
		}
		sort.Strings(c.SourceIDs)
		sort.Strings(c.Issues)
		for _, dep := range p.ProjectReferences {
			d, err := makeContext(dep.Project)
			if err != nil {
				return c, err
			}
			if dep.Context != d.InputFingerprint {
				return c, fmt.Errorf("semantic import: dependency context mismatch")
			}
			if c.Status == "complete" && d.Status != "complete" {
				return c, fmt.Errorf("semantic import: complete project has incomplete dependency")
			}
			c.DependencyContextIDs = append(c.DependencyContextIDs, d.ID)
		}
		sort.Strings(c.DependencyContextIDs)
		c.ID = c.ComputeID()
		contexts[name] = c
		visiting[name] = false
		return c, nil
	}
	for _, name := range names {
		c, err := makeContext(name)
		if err != nil {
			return nil, err
		}
		artifact.Contexts = append(artifact.Contexts, c)
	}
	symbols := map[string]semantic.Symbol{}
	occurrences := map[string]semantic.Occurrence{}
	bindings := map[string]semantic.Binding{}
	addSymbol := func(w symbolRecord) (string, error) {
		if w.Language != "csharp" || (w.NamespaceKind != "project" && w.NamespaceKind != "assembly") {
			return "", fmt.Errorf("semantic import: unsupported worker symbol namespace/language")
		}
		if w.NamespaceKind == "project" {
			found := false
			for _, name := range names {
				if w.Namespace == opts.Repo+"/"+name {
					found = true
					break
				}
			}
			if !found {
				return "", fmt.Errorf("semantic import: unknown source symbol namespace")
			}
		}
		s := semantic.Symbol{Key: semantic.SymbolKey{Language: w.Language, NamespaceKind: w.NamespaceKind, Namespace: w.Namespace, Descriptor: w.Descriptor, DescriptorKind: w.DescriptorKind}}
		s.ID = s.ComputeID()
		symbols[s.ID] = s
		return s.ID, nil
	}
	for _, r := range rows {
		if err := ctx.Err(); err != nil {
			return nil, err
		}
		source, ok := sources[r.SourcePath]
		if !ok || source.RawSHA256 != r.SourceSHA {
			return nil, fmt.Errorf("semantic import: fact source missing or mismatched")
		}
		raw := rawSources[r.SourcePath]
		off, n := r.Span.Offset, r.Span.Length
		if off > uint64(len(raw)) || n > uint64(len(raw))-off || n == 0 || string(raw[off:off+n]) != r.SourceText || !utf8.Valid(raw[off:off+n]) {
			return nil, fmt.Errorf("semantic import: source span mismatch")
		}
		c := contexts[r.Project]
		o := semantic.Occurrence{SourceID: source.ID, ContextID: c.ID, Role: r.Type, Kind: r.ReferenceKind, Offset: off, Length: n}
		o.ID = o.ComputeID()
		b := semantic.Binding{OccurrenceID: o.ID, Status: r.BindingStatus, Method: r.BindingMethod, Extractor: c.Extractor, ExtractorVersion: c.ExtractorVersion}
		if r.Symbol != nil {
			id, err := addSymbol(*r.Symbol)
			if err != nil {
				return nil, err
			}
			b.SymbolID = id
		}
		if r.EnclosingSymbol != nil {
			id, err := addSymbol(*r.EnclosingSymbol)
			if err != nil {
				return nil, err
			}
			b.EnclosingSymbolID = id
		}
		for _, candidate := range r.Candidates {
			id, err := addSymbol(candidate)
			if err != nil {
				return nil, err
			}
			b.CandidateSymbolIDs = append(b.CandidateSymbolIDs, id)
		}
		sort.Strings(b.CandidateSymbolIDs)
		if len(r.DomainFacts) > semantic.MaxDomainFacts {
			return nil, fmt.Errorf("semantic import: too many domain facts")
		}
		for _, fact := range r.DomainFacts {
			f := semantic.DomainFact{Kind: fact.Kind, Rule: fact.Rule, EvidenceScope: fact.EvidenceScope, Lifetime: fact.Lifetime, Table: fact.Table, Schema: fact.Schema}
			if len(fact.Targets) > 4 {
				return nil, fmt.Errorf("semantic import: too many domain targets")
			}
			for _, target := range fact.Targets {
				if target.Symbol == nil {
					return nil, fmt.Errorf("semantic import: missing domain target symbol")
				}
				id, err := addSymbol(*target.Symbol)
				if err != nil {
					return nil, err
				}
				f.Targets = append(f.Targets, semantic.DomainTarget{Role: target.Role, SymbolID: id})
			}
			b.DomainFacts = append(b.DomainFacts, f)
		}
		if len(r.ImplementationFacts) > semantic.MaxImplementationFacts {
			return nil, fmt.Errorf("semantic import: too many implementation facts")
		}
		for _, fact := range r.ImplementationFacts {
			if fact.InterfaceSymbol == nil || fact.ImplementingType == nil {
				return nil, fmt.Errorf("semantic import: missing implementation relationship symbol")
			}
			interfaceID, err := addSymbol(*fact.InterfaceSymbol)
			if err != nil {
				return nil, err
			}
			typeID, err := addSymbol(*fact.ImplementingType)
			if err != nil {
				return nil, err
			}
			f := semantic.ImplementationFact{Kind: fact.Kind, Rule: fact.Rule, EvidenceScope: fact.EvidenceScope, InterfaceSymbolID: interfaceID, ImplementingTypeSymbolID: typeID}
			for _, target := range []struct {
				wire *symbolRecord
				id   *string
			}{{fact.SelectedDefaultSymbol, &f.SelectedDefaultSymbolID}, {fact.DefaultTemplateSymbol, &f.DefaultTemplateSymbolID}} {
				if target.wire != nil {
					id, err := addSymbol(*target.wire)
					if err != nil {
						return nil, err
					}
					*target.id = id
				}
			}
			b.ImplementationFacts = append(b.ImplementationFacts, f)
		}
		sort.Slice(b.ImplementationFacts, func(i, j int) bool {
			return b.ImplementationFacts[i].InterfaceSymbolID+"/"+b.ImplementationFacts[i].ImplementingTypeSymbolID < b.ImplementationFacts[j].InterfaceSymbolID+"/"+b.ImplementationFacts[j].ImplementingTypeSymbolID
		})
		b.ID = b.ComputeID()
		if prior, ok := bindings[o.ID]; ok && prior.ID != b.ID {
			return nil, fmt.Errorf("semantic import: conflicting bindings at occurrence")
		}
		occurrences[o.ID] = o
		bindings[o.ID] = b
	}
	for _, s := range symbols {
		artifact.Symbols = append(artifact.Symbols, s)
	}
	for _, o := range occurrences {
		artifact.Occurrences = append(artifact.Occurrences, o)
	}
	for _, b := range bindings {
		artifact.Bindings = append(artifact.Bindings, b)
	}
	sort.Slice(artifact.Symbols, func(i, j int) bool { return artifact.Symbols[i].ID < artifact.Symbols[j].ID })
	sort.Slice(artifact.Occurrences, func(i, j int) bool { return artifact.Occurrences[i].ID < artifact.Occurrences[j].ID })
	sort.Slice(artifact.Bindings, func(i, j int) bool { return artifact.Bindings[i].ID < artifact.Bindings[j].ID })
	for _, r := range diagnosticRows {
		d := semantic.Diagnostic{ContextID: contexts[r.Project].ID, Code: r.Code, Severity: r.Severity, Message: r.Message}
		if r.SourcePath != "" {
			source, ok := sources[r.SourcePath]
			if !ok {
				return nil, fmt.Errorf("semantic import: diagnostic source missing")
			}
			raw := rawSources[r.SourcePath]
			off, n := r.Span.Offset, r.Span.Length
			if off > uint64(len(raw)) || n > uint64(len(raw))-off || !utf8.Valid(raw[:off]) || !utf8.Valid(raw[off:off+n]) {
				return nil, fmt.Errorf("semantic import: invalid diagnostic span")
			}
			d.SourceID = source.ID
			d.Offset = off
			d.Length = n
		} else if r.Span.Offset != 0 || r.Span.Length != 0 {
			return nil, fmt.Errorf("semantic import: locationless diagnostic has span")
		}
		d.ID = d.ComputeID()
		artifact.Diagnostics = append(artifact.Diagnostics, d)
	}
	sort.Slice(artifact.Diagnostics, func(i, j int) bool { return artifact.Diagnostics[i].ID < artifact.Diagnostics[j].ID })

	if err := artifact.Validate(); err != nil {
		return nil, err
	}
	return artifact, nil
}
