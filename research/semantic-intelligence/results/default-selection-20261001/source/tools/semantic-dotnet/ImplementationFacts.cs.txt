using Microsoft.CodeAnalysis;

static partial class Worker
{
    // Compiler declaration correspondence, not runtime interface dispatch.
    // Closed interface arguments and explicit default templates stay distinct.
    // Inherited class mappings and large correspondence sets remain excluded.
    static object[]? ImplementationFacts(State state, ISymbol declared)
    {
        if (state.Complete && declared is INamedTypeSymbol concrete)
            return DefaultSelections(concrete);
        if (!state.Complete || declared is not IMethodSymbol method || method.IsStatic || method.IsAbstract || method.IsExtern || method.IsGenericMethod ||
            method.MethodKind is not (MethodKind.Ordinary or MethodKind.ExplicitInterfaceImplementation) ||
            method.ContainingType is not { } type ||
            !method.Locations.Any(l => l.IsInSource)) return null;
        // Explicit default interface bodies record the compiler correspondence at
        // their own source declaration. This is an open template, not a closed
        // class implementation or an assertion that its forwarding call executes.
        if (type.TypeKind == TypeKind.Interface && method.MethodKind == MethodKind.ExplicitInterfaceImplementation &&
            method.ExplicitInterfaceImplementations.Length is > 0 and <= 32 &&
            method.ExplicitInterfaceImplementations.All(m => DomainTarget(m.ContainingType) && !m.IsGenericMethod && !m.IsStatic))
            return method.ExplicitInterfaceImplementations.Select(member => new
            {
                Kind = "interface_method_implementation", Rule = "csharp-interface-default-v1", EvidenceScope = "compile_time",
                InterfaceSymbol = Symbol(member), ImplementingType = Symbol(type)
            }).Cast<object>().ToArray();
        if (type.TypeKind != TypeKind.Class || !DomainTarget(type)) return null;
        var interfaces = new List<IMethodSymbol>();
        foreach (var iface in type.AllInterfaces)
        {
            if (!DomainTarget(iface) && !(iface.Arity > 0 && iface.ContainingType is null && iface.TypeArguments.Length <= 8 && iface.TypeArguments.All(DomainTarget))) continue;
            foreach (var member in iface.GetMembers().OfType<IMethodSymbol>())
            {
                if (member.MethodKind != MethodKind.Ordinary || member.IsStatic || !member.IsAbstract || member.IsGenericMethod) continue;
                if (!SymbolEqualityComparer.Default.Equals(type.FindImplementationForInterfaceMember(member), method)) continue;
                interfaces.Add(member);
                if (interfaces.Count > 32) return null;
            }
        }
        if (interfaces.Count == 0) return null;
        return interfaces.Select(member => new
        {
            Kind = "interface_method_implementation", Rule = ClosedInterfaceMethod(member) ? "csharp-interface-closed-v1" : "csharp-interface-v1", EvidenceScope = "compile_time",
            InterfaceSymbol = Symbol(member), ImplementingType = Symbol(type)
        }).OrderBy(fact => System.Text.Json.JsonSerializer.Serialize(fact.InterfaceSymbol, Json), StringComparer.Ordinal).Cast<object>().ToArray();
    }

    // Anchor selection at the concrete class, not at the shared open body.
    // Roslyn resolves overrides/most-specific implementations; no name matching.
    static object[]? DefaultSelections(INamedTypeSymbol type)
    {
        if (type.TypeKind != TypeKind.Class || type.IsAbstract || !DomainTarget(type) ||
            !type.Locations.Any(l => l.IsInSource)) return null;
        var facts = new List<object>();
        foreach (var iface in type.AllInterfaces.Where(DomainTarget))
        foreach (var member in iface.GetMembers().OfType<IMethodSymbol>())
        {
            if (member.MethodKind != MethodKind.Ordinary || member.IsStatic || !member.IsAbstract || member.IsGenericMethod) continue;
            if (type.FindImplementationForInterfaceMember(member) is not IMethodSymbol selected ||
                !ClosedInterfaceMethod(selected) || selected.IsAbstract || selected.IsStatic || selected.IsExtern ||
                selected.MethodKind != MethodKind.ExplicitInterfaceImplementation ||
                !selected.ExplicitInterfaceImplementations.Any(m => SymbolEqualityComparer.Default.Equals(m, member)) ||
                !selected.Locations.Any(l => l.IsInSource)) continue;
            facts.Add(new {
                Kind = "interface_default_selection", Rule = "csharp-interface-selection-v1", EvidenceScope = "compile_time",
                InterfaceSymbol = Symbol(member), ImplementingType = Symbol(type),
                SelectedDefaultSymbol = Symbol(selected), DefaultTemplateSymbol = Symbol(selected.OriginalDefinition)
            });
            if (facts.Count > 32) return null;
        }
        return facts.Count == 0 ? null : facts.OrderBy(f => System.Text.Json.JsonSerializer.Serialize(f, Json), StringComparer.Ordinal).ToArray();
    }
}
