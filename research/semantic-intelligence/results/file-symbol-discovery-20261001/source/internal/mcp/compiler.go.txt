// CompilerReader returns owned, bounded records, never views into a mapping.
type CompilerReader interface {
	Bindings(context.Context, semanticindex.Position, int) (semanticindex.QueryResult, error)
	Definitions(context.Context, string, semanticindex.Filter, int) (semanticindex.QueryResult, error)
	SourceContexts(context.Context, string, string, int) ([]semanticindex.SourceContext, bool, error)
	SourceContext(context.Context, string, string, string) (semanticindex.SourceContext, bool, error)
}
type CompilerSession struct {
	Reader                                        CompilerReader
	SnapshotID, ArtifactSHA256, CorpusFingerprint string
	Release                                       func()
}
type CompilerSessionProvider interface {
	AcquireCompilerSession(context.Context) (CompilerSession, error)
}

// CompilerTools has a stable catalog even when the current generation has no
// query index. Every invocation acquires its own immutable serving lease.
func CompilerTools(provider CompilerSessionProvider) []ToolHandler {
	return []ToolHandler{&compilerTool{provider, "compiler_binding_at"}, &compilerTool{provider, "compiler_definitions"}, &compilerContractImpactTool{provider: provider}, &compilerContractContextTool{provider: provider}, &compilerContractPathsTool{provider: provider}, &compilerImplementationsTool{provider: provider}, &compilerSymbolsTool{provider: provider}}
}

type compilerTool struct {
	provider CompilerSessionProvider
	name     string
}

func (t *compilerTool) Name() string { return t.name }

type CompilerContext struct {
	Generated bool   `json:"generated"`
	ContextID string `json:"context_id"`
	Project   string `json:"project"`
	Repo      string `json:"repo"`
	Path      string `json:"path"`
	RawSHA256 string `json:"raw_sha256"`
}
type CompilerSymbol struct {
	ID             string `json:"id"`
	Language       string `json:"language"`
	NamespaceKind  string `json:"namespace_kind"`
	Namespace      string `json:"namespace"`
	Descriptor     string `json:"descriptor"`
	DescriptorKind string `json:"descriptor_kind"`
}

// CompilerDomainFact records framework API/type observations in a selected
// compiler context. It does not assert runtime delivery, resolution, or storage.
type CompilerDomainTarget struct {
	Role   string         `json:"role"`
	Symbol CompilerSymbol `json:"symbol"`
}
type CompilerDomainFact struct {
	Kind          string                 `json:"kind"`
	Rule          string                 `json:"rule"`
	EvidenceScope string                 `json:"evidence_scope"`
	Targets       []CompilerDomainTarget `json:"targets"`
	Lifetime      string                 `json:"lifetime,omitempty"`
	Table         string                 `json:"table,omitempty"`
	Schema        string                 `json:"schema,omitempty"`
}
type CompilerBinding struct {
	Generated             bool                          `json:"generated"`
	OccurrenceID          string                        `json:"occurrence_id"`
	BindingStatus         string                        `json:"binding_status"`
	ReferenceKind         string                        `json:"reference_kind"`
	Role                  string                        `json:"role"`
	Repo                  string                        `json:"repo"`
	Path                  string                        `json:"path"`
	RawSHA256             string                        `json:"raw_sha256"`
	SourceID              string                        `json:"source_id"`
	SourceSnapshotID      string                        `json:"source_snapshot_id"`
	ContextID             string                        `json:"context_id"`
	Project               string                        `json:"project"`
	ByteOffset            uint64                        `json:"byte_offset"`
	ByteLength            uint64                        `json:"byte_length"`
	Symbol                *CompilerSymbol               `json:"symbol,omitempty"`
	Candidates            []CompilerSymbol              `json:"candidates"`
	EnclosingSymbolID     string                        `json:"enclosing_symbol_id,omitempty"`
	Method                string                        `json:"method"`
	Extractor             string                        `json:"extractor"`
	ExtractorVersion      string                        `json:"extractor_version"`
	DomainFacts           []CompilerDomainFact          `json:"domain_facts,omitempty"`
	ImplementationFacts   []semantic.ImplementationFact `json:"implementation_facts,omitempty"`
	ImplementationSymbols []CompilerSymbol              `json:"implementation_symbols,omitempty"`
}
type CompilerResult struct {
	Status         string            `json:"status"`
	Evidence       string            `json:"evidence"`
	SnapshotID     string            `json:"snapshot_id"`
	ArtifactSHA256 string            `json:"artifact_sha256"`
	Results        []CompilerBinding `json:"results"`
	Contexts       []CompilerContext `json:"contexts"`
	Truncated      bool              `json:"truncated"`
	Error          *compilerError    `json:"error,omitempty"`
}
type compilerError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

func compilerSymbol(s semantic.Symbol) CompilerSymbol {
	return CompilerSymbol{s.ID, s.Key.Language, s.Key.NamespaceKind, s.Key.Namespace, s.Key.Descriptor, s.Key.DescriptorKind}
}
func compilerBinding(r semanticindex.Result) (CompilerBinding, error) {
	out := CompilerBinding{Generated: r.Source.Generated, OccurrenceID: r.Occurrence.ID, BindingStatus: r.Binding.Status, ReferenceKind: r.Occurrence.Kind, Role: r.Occurrence.Role, Repo: r.Snapshot.Repo, Path: r.Source.Path, RawSHA256: r.Source.RawSHA256, SourceID: r.Source.ID, SourceSnapshotID: r.Snapshot.ID, ContextID: r.Context.ID, Project: r.Context.Project, ByteOffset: r.Occurrence.Offset, ByteLength: r.Occurrence.Length, Candidates: []CompilerSymbol{}, EnclosingSymbolID: r.Binding.EnclosingSymbolID, Method: r.Binding.Method, Extractor: r.Binding.Extractor, ExtractorVersion: r.Binding.ExtractorVersion}
	if r.Symbol != nil {
		s := compilerSymbol(*r.Symbol)
		out.Symbol = &s
	}
	for _, s := range r.Candidates {
		out.Candidates = append(out.Candidates, compilerSymbol(s))
	}
	domainSymbols := make(map[string]semantic.Symbol, len(r.DomainSymbols))
	for _, symbol := range r.DomainSymbols {
		domainSymbols[symbol.ID] = symbol
	}
	for _, fact := range r.Binding.DomainFacts {
		value := CompilerDomainFact{Kind: fact.Kind, Rule: fact.Rule, EvidenceScope: fact.EvidenceScope, Lifetime: fact.Lifetime, Table: fact.Table, Schema: fact.Schema, Targets: []CompilerDomainTarget{}}
		for _, target := range fact.Targets {
			value.Targets = append(value.Targets, CompilerDomainTarget{Role: target.Role, Symbol: compilerSymbol(domainSymbols[target.SymbolID])})
		}
		out.DomainFacts = append(out.DomainFacts, value)
	}
	if err := compilerImplementationEvidence(r, &out); err != nil {
		return CompilerBinding{}, err
	}
	return out, nil
}
func compilerRelative(s string) bool {
	if s == "" || len(s) > 4096 || strings.Contains(s, "\\") || strings.HasPrefix(s, "/") {
		return false
	}
	for _, p := range strings.Split(s, "/") {
		if p == "" || p == "." || p == ".." {
			return false
		}
	}
	return !strings.ContainsAny(s, "\x00\r\n")
}
func compilerID(s string) bool {
	return len(s) > 0 && len(s) <= 256 && !strings.ContainsAny(s, "\x00\r\n")
}
func (t *compilerTool) Call(ctx context.Context, raw json.RawMessage) (map[string]interface{}, error) {
	out := CompilerResult{Status: "semantic_unavailable", Evidence: "recorded-compiler-context", Results: []CompilerBinding{}, Contexts: []CompilerContext{}}
	respond := func(message string, bad bool) map[string]interface{} {
		return StructuredResultWithSnapshot(message, out, bad, SnapshotIdentity{Cacheable: false})
	}
	invalid := func(message string) (map[string]interface{}, error) {
		out.Status = "invalid_arguments"
		out.Error = &compilerError{"invalid_arguments", message}
		return respond(message, true), nil
	}
	var args struct {
		Repo    string  `json:"repo"`
		Path    string  `json:"path"`
		Offset  *uint64 `json:"byte_offset"`
		Context string  `json:"context_id"`
		RawSHA  string  `json:"raw_sha256"`
		Symbol  string  `json:"symbol_id"`
		Limit   *int    `json:"limit"`
	}
	dec := json.NewDecoder(bytes.NewReader(raw))
	dec.DisallowUnknownFields()
	if err := dec.Decode(&args); err != nil {
		return invalid("invalid compiler tool arguments")
	}
	var extra any
	if dec.Decode(&extra) != io.EOF {
		return invalid("arguments must be one object")
	}
	var fields map[string]json.RawMessage
	if err := json.Unmarshal(raw, &fields); err != nil || fields == nil {
		return invalid("arguments must be an object")
	}
	for name := range fields {
		allowed := name == "repo" || name == "context_id" || name == "limit"
		if t.name == "compiler_binding_at" {
			allowed = allowed || name == "path" || name == "byte_offset" || name == "raw_sha256"
		} else {
			allowed = allowed || name == "symbol_id"
		}
		if !allowed {
			return invalid("unsupported argument for compiler tool")
		}
	}
	limit := semanticindex.MaxResults
	if args.Limit != nil {
		limit = *args.Limit
	}
	if limit < 1 || limit > semanticindex.MaxResults {
		return invalid("limit must be 1..100")
	}
	if args.Context != "" && !compilerID(args.Context) {
		return invalid("invalid context_id")
	}
	if t.name == "compiler_binding_at" {
		if !compilerRelative(args.Repo) || !compilerRelative(args.Path) || args.Offset == nil || args.Symbol != "" {
			return invalid("repo, path and exact raw byte_offset are required")
		}
		if args.Context != "" {
			b, e := hex.DecodeString(args.RawSHA)
			if e != nil || len(b) != 32 || strings.ToLower(args.RawSHA) != args.RawSHA {
				return invalid("context_id requires lowercase raw_sha256")
			}
		}
	} else if !compilerID(args.Symbol) || (args.Repo != "" && !compilerRelative(args.Repo)) || args.Path != "" || args.Offset != nil || args.RawSHA != "" {
		return invalid("symbol_id and optional repo/context_id are required")
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	if t.provider == nil {
		return respond("Compiler evidence unavailable in this snapshot.", false), nil
	}
	session, err := t.provider.AcquireCompilerSession(ctx)
	if err != nil {
		return nil, err
	}
	if session.Release != nil {
		defer session.Release()
	}
	out.SnapshotID, out.ArtifactSHA256 = session.SnapshotID, session.ArtifactSHA256
	if session.Reader == nil {
		return respond("Compiler evidence unavailable in this snapshot.", false), nil
	}
	var query semanticindex.QueryResult
	if t.name == "compiler_binding_at" {
		if args.Context == "" {
			contexts, truncated, e := session.Reader.SourceContexts(ctx, args.Repo, args.Path, limit)
			if e != nil {
				return nil, e
			}
			if len(contexts) == 0 {
				out.Status = "source_not_found"
				return respond("No recorded compiler context covers this repository/path.", false), nil
			}
			out.Status = "context_required"
			out.Truncated = truncated
			for _, c := range contexts {
				out.Contexts = append(out.Contexts, CompilerContext{Generated: c.Source.Generated, ContextID: c.Context.ID, Project: c.Context.Project, Repo: c.Snapshot.Repo, Path: c.Source.Path, RawSHA256: c.Source.RawSHA256})
			}
			return respond("Select a recorded context and raw_sha256 before querying a binding.", false), nil
		}
		source, found, e := session.Reader.SourceContext(ctx, args.Repo, args.Path, args.Context)
		if e != nil {
			return nil, e
		}
		if !found {
			out.Status = "context_not_found"
			return respond("Recorded source/context is not available in this snapshot.", false), nil
		}
		if source.Source.RawSHA256 != args.RawSHA {
			out.Status = "source_mismatch"
			return respond("Raw source hash does not match the recorded context.", true), nil
		}
		query, err = session.Reader.Bindings(ctx, semanticindex.Position{Repo: args.Repo, Path: args.Path, Offset: *args.Offset, BuildContextID: args.Context}, limit)
		if err != nil {
			return nil, err
		}
		for _, r := range query.Results {
			if r.Source.RawSHA256 != args.RawSHA {
				out.Status = "source_mismatch"
				return respond("Raw source hash does not match the recorded context.", true), nil
			}
		}
	} else {
		query, err = session.Reader.Definitions(ctx, args.Symbol, semanticindex.Filter{Repo: args.Repo, BuildContextID: args.Context}, limit)
		if err != nil {
			return nil, err
		}
	}
	out.Status = "ok"
	out.Truncated = query.Truncated
	for _, r := range query.Results {
		binding, err := compilerBinding(r)
		if err != nil {
			return nil, err
		}
		out.Results = append(out.Results, binding)
	}
	if len(out.Results) == 0 {
		out.Status = "no_binding"
		if t.name == "compiler_definitions" {
			out.Status = "no_definition"
		}
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	return respond(fmt.Sprintf("%d recorded compiler facts; evidence is historical, not runtime dispatch.", len(out.Results)), false), nil
}
func (t *compilerTool) Specification() ToolSpecification {
	props := map[string]interface{}{"repo": stringSchema(), "context_id": stringSchema(), "limit": map[string]interface{}{"type": "integer", "minimum": 1, "maximum": 100}}
	required := []string{"symbol_id"}
	description := "Return recorded compiler declaration variants for a qualified symbol; optional repo/context filters. Optional implementation_facts link source method declarations to compiler-verified interface members via implementation_symbols; this is historical compiler evidence, not runtime dispatch or reverse implementation discovery."
	if t.name == "compiler_binding_at" {
		required = []string{"repo", "path", "byte_offset"}
		props["path"] = stringSchema()
		props["byte_offset"] = map[string]interface{}{"type": "integer", "minimum": 0}
		props["raw_sha256"] = stringSchema()
		description = "Look up compiler bindings at an exact raw UTF8 token-start byte offset (including BOM). Omit context_id to list recorded contexts; selecting one requires its raw_sha256. read_source returns indexed text with a leading BOM removed: verify complete text with or without the UTF8 BOM against raw_sha256 before computing raw offsets; stop if neither matches. Compiler evidence is historical and project-specific. Optional implementation_facts on method declarations identify compiler interface correspondence via implementation_symbols, not runtime dispatch. Optional domain_facts describe compile-time framework observations, not runtime delivery, service resolution, or database activity."
	} else {
		props["symbol_id"] = stringSchema()
	}
	_, binding, _, choices := compilerEvidenceSchemas()
	output := objectSchema([]string{"status", "evidence", "snapshot_id", "artifact_sha256", "results", "contexts", "truncated"}, map[string]interface{}{"status": stringSchema(), "evidence": stringSchema(), "snapshot_id": stringSchema(), "artifact_sha256": stringSchema(), "results": arrayOf(binding), "contexts": arrayOf(choices), "truncated": boolSchema(), "error": errorPropertySchema()})
	return ToolSpecification{Name: t.name, Description: description, InputSchema: objectSchema(required, props), OutputSchema: output}
}

func compilerEvidenceSchemas() (symbols, binding, domainFact, choices map[string]interface{}) {
	symbols = objectSchema([]string{"id", "language", "namespace_kind", "namespace", "descriptor", "descriptor_kind"}, map[string]interface{}{"id": stringSchema(), "language": stringSchema(), "namespace_kind": stringSchema(), "namespace": stringSchema(), "descriptor": stringSchema(), "descriptor_kind": stringSchema()})
	facts := map[string]interface{}{}
	for _, k := range []string{"occurrence_id", "binding_status", "reference_kind", "role", "repo", "path", "raw_sha256", "source_id", "source_snapshot_id", "context_id", "project", "enclosing_symbol_id", "method", "extractor", "extractor_version"} {
		facts[k] = stringSchema()
	}
	facts["generated"] = boolSchema()
	facts["byte_offset"] = intSchema()
	facts["byte_length"] = intSchema()
	facts["symbol"] = symbols
	facts["candidates"] = arrayOf(symbols)
	domainTarget := objectSchema([]string{"role", "symbol"}, map[string]interface{}{"role": stringSchema(), "symbol": symbols})
	domainFact = objectSchema([]string{"kind", "rule", "evidence_scope", "targets"}, map[string]interface{}{"kind": stringSchema(), "rule": stringSchema(), "evidence_scope": stringSchema(), "targets": arrayOf(domainTarget), "lifetime": stringSchema(), "table": stringSchema(), "schema": stringSchema()})
	facts["domain_facts"] = arrayOf(domainFact)
	implementation := objectSchema([]string{"kind", "rule", "evidence_scope", "interface_symbol_id", "implementing_type_symbol_id"}, map[string]interface{}{"kind": stringSchema(), "rule": stringSchema(), "evidence_scope": stringSchema(), "interface_symbol_id": stringSchema(), "implementing_type_symbol_id": stringSchema()})
	implementationFacts := arrayOf(implementation)
	implementationFacts["maxItems"] = semantic.MaxImplementationFacts
	implementationSymbols := arrayOf(symbols)
	implementationSymbols["maxItems"] = semantic.MaxImplementationFacts * 2
	facts["implementation_facts"] = implementationFacts
	facts["implementation_symbols"] = implementationSymbols
	choices = objectSchema([]string{"context_id", "project", "repo", "path", "raw_sha256"}, map[string]interface{}{"context_id": stringSchema(), "project": stringSchema(), "repo": stringSchema(), "path": stringSchema(), "raw_sha256": stringSchema(), "generated": boolSchema()})
	return symbols, objectSchema(nil, facts), domainFact, choices
}
