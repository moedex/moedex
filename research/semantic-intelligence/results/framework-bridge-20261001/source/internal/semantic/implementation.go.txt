package semantic

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"strconv"
	"strings"
)

// ImplementationFact records the compiler's interface-member implementation
// relationship at a source method declaration. It does not prove dispatch.
type ImplementationFact struct {
	Kind                     string `json:"kind"`
	Rule                     string `json:"rule"`
	EvidenceScope            string `json:"evidence_scope"`
	InterfaceSymbolID        string `json:"interface_symbol_id"`
	ImplementingTypeSymbolID string `json:"implementing_type_symbol_id"`
}

const MaxImplementationFacts = 32

func ValidateImplementationFacts(facts []ImplementationFact) error {
	if len(facts) > MaxImplementationFacts {
		return fmt.Errorf("semantic: too many implementation facts")
	}
	last := ""
	for _, f := range facts {
		if f.Kind != "interface_method_implementation" || !oneOf(f.Rule, "csharp-interface-v1", "csharp-interface-closed-v1", "csharp-interface-default-v1") || f.EvidenceScope != "compile_time" {
			return fmt.Errorf("semantic: unsupported implementation evidence")
		}
		for _, id := range []string{f.InterfaceSymbolID, f.ImplementingTypeSymbolID} {
			if !strings.HasPrefix(id, "symbol:") || !digest(strings.TrimPrefix(id, "symbol:")) {
				return fmt.Errorf("semantic: invalid implementation target")
			}
		}
		key := f.InterfaceSymbolID + "/" + f.ImplementingTypeSymbolID
		if key <= last {
			return fmt.Errorf("semantic: implementation facts must be sorted and unique")
		}
		last = key
	}
	return nil
}

// ValidateImplementationSymbols checks bounded identity shape and the source
// method/type relationship. Compiler provenance establishes interface semantics.
func ValidateImplementationSymbols(f ImplementationFact, implementation, iface, typ Symbol) error {
	method := func(s Symbol) bool {
		k := s.Key
		name, _, _ := strings.Cut(k.Descriptor, "(")
		return k.Language == "csharp" && (k.NamespaceKind == "project" || k.NamespaceKind == "assembly") && k.DescriptorKind == "documentation_comment_id" && strings.HasPrefix(name, "M:") && !strings.Contains(name, "`") && !strings.Contains(name, ".#")
	}
	validImplementation, validInterface, validType := method(implementation), method(iface), ValidateDomainTarget(typ) == nil
	if f.Rule == "csharp-interface-closed-v1" {
		validInterface = ValidateConstructedInterfaceMethod(iface.Key) == nil
	}
	if f.Rule == "csharp-interface-default-v1" {
		validImplementation = implementation.Key.Language == "csharp" && implementation.Key.DescriptorKind == "documentation_comment_id" && strings.HasPrefix(implementation.Key.Descriptor, "M:") && strings.Contains(implementation.Key.Descriptor, "#")
		validType = typ.Key.Language == "csharp" && typ.Key.DescriptorKind == "documentation_comment_id" && strings.HasPrefix(typ.Key.Descriptor, "T:") && !strings.ContainsAny(typ.Key.Descriptor, "{}[]")
	}
	if !validImplementation || implementation.Key.NamespaceKind != "project" || !validInterface || !validType || typ.Key.NamespaceKind != "project" || f.InterfaceSymbolID != iface.ID || f.ImplementingTypeSymbolID != typ.ID || implementation.ID == iface.ID {
		return fmt.Errorf("semantic: invalid implementation symbols")
	}
	name, _, _ := strings.Cut(implementation.Key.Descriptor, "(")
	member, matchesType := strings.CutPrefix(name, "M:"+strings.TrimPrefix(typ.Key.Descriptor, "T:")+".")
	if implementation.Key.Namespace != typ.Key.Namespace || !matchesType || member == "" || strings.Contains(member, ".") {
		return fmt.Errorf("semantic: implementation method is outside implementing type")
	}
	return nil
}

// ValidateConstructedInterfaceMethod checks qualified closed argument identities.
// Definition is the original documentation ID; the outer namespace owns it.
func ValidateConstructedInterfaceMethod(k SymbolKey) error {
	bad := func() error { return fmt.Errorf("semantic: invalid constructed interface member") }
	if k.Language != "csharp" || !oneOf(k.NamespaceKind, "project", "assembly") || k.Namespace == "" || k.DescriptorKind != "constructed_interface_method_v1" || len(k.Descriptor) > 32768 {
		return bad()
	}
	var value struct {
		Definition string      `json:"definition"`
		Arguments  []SymbolKey `json:"arguments"`
	}
	d := json.NewDecoder(bytes.NewBufferString(k.Descriptor))
	d.DisallowUnknownFields()
	if d.Decode(&value) != nil {
		return bad()
	}
	var extra any
	if d.Decode(&extra) != io.EOF {
		return bad()
	}
	name, _, _ := strings.Cut(value.Definition, "(")
	if !strings.HasPrefix(name, "M:") || !strings.Contains(name, "`") || strings.Contains(name, "``") || len(value.Arguments) < 1 || len(value.Arguments) > 8 {
		return bad()
	}
	_, member, ok := strings.Cut(name, "`")
	arityText, _, dot := strings.Cut(member, ".")
	arity, err := strconv.Atoi(arityText)
	if !ok || !dot || err != nil || arity != len(value.Arguments) || strings.Count(name, "`") != 1 {
		return bad()
	}
	for _, a := range value.Arguments {
		if !oneOf(a.NamespaceKind, "project", "assembly") || !required(a.Namespace) || !required(a.Descriptor) || ValidateDomainTarget(Symbol{Key: a}) != nil {
			return bad()
		}
	}
	return nil
}

func ValidateImplementationExtractor(f ImplementationFact, version string) error {
	if f.Rule == "csharp-interface-v1" && (version == "6" || version == "7") || (f.Rule == "csharp-interface-closed-v1" || f.Rule == "csharp-interface-default-v1") && version == "7" {
		return nil
	}
	return fmt.Errorf("semantic: implementation rule requires supported extractor version")
}
