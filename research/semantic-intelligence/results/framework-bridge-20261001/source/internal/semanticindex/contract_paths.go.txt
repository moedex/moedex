package semanticindex

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"moedex/internal/semantic"
	"sort"
	"strings"
)

type ContractPath struct {
	Seed                    ContractImpactMatch `json:"seed"`
	Caller                  Result              `json:"caller"`
	CallerOwner             semantic.Symbol     `json:"caller_owner"`
	Implementation          *Result             `json:"implementation,omitempty"`
	ImplementationFactIndex int                 `json:"implementation_fact_index"`
}
type ContractPathsResult struct {
	Supported      bool                  `json:"supported"`
	Contract       *semantic.Symbol      `json:"contract,omitempty"`
	Seeds          []ContractImpactMatch `json:"seeds"`
	Paths          []ContractPath        `json:"paths"`
	SeedsTruncated bool                  `json:"seeds_truncated"`
	PathsTruncated bool                  `json:"paths_truncated"`
	Records        int                   `json:"records"`
	RowsVisited    int                   `json:"rows_visited"`
	WorkLimit      int                   `json:"work_limit"`
}

// ContractPaths returns one exact caller step, optionally through a recorded
// interface implementation. It establishes candidate static paths, not dispatch.
// Contexts must be explicit; a relationship uses the publisher's own compilation.
func (x *Index) ContractPaths(ctx context.Context, symbolID string, contextIDs []string, limit int) (ContractPathsResult, error) {
	out := ContractPathsResult{WorkLimit: ContractWorkLimit}
	if err := x.check(ctx, limit); err != nil {
		return out, err
	}
	if !strings.HasPrefix(symbolID, "symbol:") || !hexBytes([]byte(strings.TrimPrefix(symbolID, "symbol:")), 64) {
		return out, fmt.Errorf("semanticindex: exact symbol ID required")
	}
	if len(contextIDs) == 0 {
		return out, fmt.Errorf("%w: explicit contexts required", ErrContractContextSelection)
	}
	selected, err := x.contractContexts(contextIDs)
	if err != nil {
		return out, err
	}
	if x.version < 4 {
		return out, nil
	}
	out.Supported = true
	sym, ok := x.domainSymbolRow(symbolID)
	if !ok {
		return out, nil
	}
	cost := 2 * x.rowCost(symbols, sym, 0, 1, 2, 3, 4, 5)
	charge := func(n uint64) error {
		if n > MaxQueryBytes || cost > MaxQueryBytes-n {
			return fmt.Errorf("semanticindex: query result exceeds byte budget")
		}
		cost += n
		return nil
	}
	if cost > MaxQueryBytes {
		return ContractPathsResult{}, fmt.Errorf("semanticindex: query result exceeds byte budget")
	}
	contract := x.symbol(sym)
	out.Contract = &contract
	visit := func() bool {
		if out.RowsVisited >= ContractWorkLimit {
			return false
		}
		out.RowsVisited++
		return true
	}
	type seedRow struct {
		fact, domain, owner uint64
		match               ContractImpactMatch
	}
	var seeds []seedRow
	seedLimit := 20
	if limit < seedLimit {
		seedLimit = limit
	}
	seedDone := false
	for _, c := range selected {
		if seedDone {
			break
		}
		for row := x.contractStart(sym, c); row < x.sections[contractPostings].n; row++ {
			if err := ctx.Err(); err != nil {
				return ContractPathsResult{}, err
			}
			p := x.contractPosting(row)
			if p[0] != sym || x.field(facts, p[1], 3) != c {
				break
			}
			if !visit() {
				out.SeedsTruncated = true
				seedDone = true
				break
			}
			var fs []semantic.DomainFact
			_ = json.Unmarshal(x.domainPayload(p[1]), &fs)
			f := fs[p[2]]
			owner := x.field(facts, p[1], 10)
			if f.Kind != "message_publish" || owner == 0 || x.text(x.field(facts, p[1], 7)) != "invocation" {
				continue
			}
			if !bytes.HasPrefix(x.textBytes(x.field(symbols, owner-1, 4)), []byte("M:")) {
				continue
			}
			if len(seeds) == seedLimit {
				out.SeedsTruncated = true
				seedDone = true
				break
			}
			n := x.resultCost(p[1]) + 2*x.rowCost(symbols, owner-1, 0, 1, 2, 3, 4, 5) + 512
			if err := charge(n); err != nil {
				return ContractPathsResult{}, err
			}
			ownerSymbol := x.symbol(owner - 1)
			match := ContractImpactMatch{Result: x.result(p[1]), DomainFactIndex: int(p[2]), TargetRoles: []string{"message"}, Owner: &ownerSymbol}
			seeds = append(seeds, seedRow{p[1], p[2], owner - 1, match})
			out.Seeds = append(out.Seeds, match)
			out.Records++
		}
	}
	stopped := false
	for _, seed := range seeds {
		if stopped {
			break
		}
		callers := func(target uint64, implementationFact *uint64, ordinal int) error {
			for _, c := range selected {
				start := sort.Search(int(x.sections[invocationPostings].n), func(j int) bool {
					k := x.invocationKey(x.field(invocationPostings, uint64(j), 0))
					return k[0] > target+1 || k[0] == target+1 && k[1] >= c
				})
				for row := uint64(start); row < x.sections[invocationPostings].n; row++ {
					if err := ctx.Err(); err != nil {
						return err
					}
					f := x.field(invocationPostings, row, 0)
					key := x.invocationKey(f)
					if key[0] != target+1 || key[1] != c {
						break
					}
					if !visit() {
						out.PathsTruncated = true
						stopped = true
						return nil
					}
					records := 2
					if implementationFact != nil {
						records++
					}
					if out.Records+records > limit {
						out.PathsTruncated = true
						stopped = true
						return nil
					}
					owner := x.field(facts, f, 10) - 1
					n := x.resultCost(seed.fact) + 2*x.rowCost(symbols, seed.owner, 0, 1, 2, 3, 4, 5) + x.resultCost(f) + 2*x.rowCost(symbols, owner, 0, 1, 2, 3, 4, 5) + 1024
					if implementationFact != nil {
						n += x.resultCost(*implementationFact)
					}
					if err := charge(n); err != nil {
						return err
					}
					path := ContractPath{Seed: seed.match, Caller: x.result(f), CallerOwner: x.symbol(owner), ImplementationFactIndex: -1}
					if implementationFact != nil {
						r := x.result(*implementationFact)
						path.Implementation = &r
						path.ImplementationFactIndex = ordinal
					}
					out.Paths = append(out.Paths, path)
					out.Records += records
				}
			}
			return nil
		}
		if err := callers(seed.owner, nil, -1); err != nil {
			return ContractPathsResult{}, err
		}
		if stopped {
			break
		}
		c := x.field(facts, seed.fact, 3)
		start := sort.Search(int(x.sections[implementationPostings].n), func(j int) bool {
			k := x.implementationKey(x.implementationPosting(uint64(j)))
			return k[0] > seed.owner || k[0] == seed.owner && k[1] >= c
		})
		for row := uint64(start); row < x.sections[implementationPostings].n; row++ {
			if err := ctx.Err(); err != nil {
				return ContractPathsResult{}, err
			}
			p := x.implementationPosting(row)
			k := x.implementationKey(p)
			if k[0] != seed.owner || k[1] != c {
				break
			}
			if !visit() {
				out.PathsTruncated = true
				stopped = true
				break
			}
			var fs []semantic.ImplementationFact
			_ = json.Unmarshal(x.implementationPayload(p[1]), &fs)
			// A default body is an interface template, not a selected closed class.
			// Reverse lookup exposes it, but automatic wrapper paths cannot discharge
			// the additional dispatch and type-substitution obligations.
			if fs[p[2]].Rule == "csharp-interface-default-v1" {
				continue
			}
			target, _ := x.domainSymbolRow(fs[p[2]].InterfaceSymbolID)
			if err := callers(target, &p[1], int(p[2])); err != nil {
				return ContractPathsResult{}, err
			}
			if stopped {
				break
			}
		}
	}
	if out.SeedsTruncated {
		out.PathsTruncated = true
	}
	if err := ctx.Err(); err != nil {
		return ContractPathsResult{}, err
	}
	return out, nil
}
