package semantic

import (
	"strings"
	"testing"
)

func TestEndpointPostFact(t *testing.T) {
	a := domainFixture()
	handler := a.Symbols[0]
	handler.Key.Descriptor = "M:Routes.Submit(System.Int32)"
	handler.ID = handler.ComputeID()
	api := a.Symbols[1]
	api.Key.Namespace = "Microsoft.AspNetCore.Routing, Version=8.0.0.0"
	api.Key.Descriptor = "M:Microsoft.AspNetCore.Builder.EndpointRouteBuilderExtensions.MapPost(Microsoft.AspNetCore.Routing.IEndpointRouteBuilder,System.String,System.Delegate)"
	api.ID = api.ComputeID()
	f := DomainFact{Kind: "endpoint_post_configuration", Rule: "csharp-endpoint-v1", EvidenceScope: "compile_time", RoutePattern: "/order/{id}", Targets: []DomainTarget{{Role: "handler", SymbolID: handler.ID}}}
	if err := ValidateDomainFacts([]DomainFact{f}); err != nil {
		t.Fatal(err)
	}
	if err := ValidateDomainAPI(f, api); err != nil {
		t.Fatal(err)
	}
	if err := ValidateDomainFactTarget(f, "handler", handler); err != nil {
		t.Fatal(err)
	}
	for _, mutate := range []func(*DomainFact){
		func(f *DomainFact) { f.RoutePattern = "" }, func(f *DomainFact) { f.RoutePattern = " " }, func(f *DomainFact) { f.RoutePattern = "/\x00" }, func(f *DomainFact) { f.RoutePattern = string([]byte{255}) }, func(f *DomainFact) { f.RoutePattern = strings.Repeat("x", 1025) }, func(f *DomainFact) { f.Kind = "message_response" }, func(f *DomainFact) { f.Rule = "csharp-framework-v6" }, func(f *DomainFact) { f.EvidenceScope = "runtime" }, func(f *DomainFact) { f.Lifetime = "singleton" }, func(f *DomainFact) { f.Table = "x" }, func(f *DomainFact) { f.Targets[0].Role = "service" },
	} {
		bad := f
		bad.Targets = append([]DomainTarget(nil), f.Targets...)
		mutate(&bad)
		if ValidateDomainFacts([]DomainFact{bad}) == nil {
			t.Fatalf("invalid fact accepted %+v", bad)
		}
	}
	for _, mutate := range []func(*Symbol){
		func(s *Symbol) { s.Key.NamespaceKind = "project" }, func(s *Symbol) { s.Key.Namespace = "Lookalike" }, func(s *Symbol) { s.Key.Descriptor = strings.ReplaceAll(s.Key.Descriptor, "MapPost", "MapGet") }, func(s *Symbol) {
			s.Key.Descriptor = strings.ReplaceAll(s.Key.Descriptor, "System.Delegate", "Microsoft.AspNetCore.Http.RequestDelegate")
		},
	} {
		bad := api
		mutate(&bad)
		if ValidateDomainAPI(f, bad) == nil {
			t.Fatalf("wrong API accepted %+v", bad)
		}
	}
	for _, descriptor := range []string{"T:Routes", "M:Routes`1.Submit", "M:Routes.Submit``1", "M:Routes.#ctor", "M:"} {
		bad := handler
		bad.Key.Descriptor = descriptor
		if ValidateDomainFactTarget(f, "handler", bad) == nil {
			t.Fatal(descriptor)
		}
	}
	bad := handler
	bad.Key.NamespaceKind = "assembly"
	if ValidateDomainFactTarget(f, "handler", bad) == nil {
		t.Fatal("metadata handler accepted")
	}
	legacy := domainFixture().Bindings[0].DomainFacts[0]
	legacy.RoutePattern = "/unexpected"
	if ValidateDomainFacts([]DomainFact{legacy}) == nil {
		t.Fatal("pattern on old fact accepted")
	}
	for _, version := range []string{"11", "12", "13", "14", "15"} {
		a.Symbols[0] = handler
		a.Symbols[1] = api
		c := &a.Contexts[0]
		c.ExtractorVersion = version
		c.ID = c.ComputeID()
		o := &a.Occurrences[0]
		o.ContextID = c.ID
		o.ID = o.ComputeID()
		b := &a.Bindings[0]
		b.ExtractorVersion = version
		b.OccurrenceID = o.ID
		b.SymbolID = api.ID
		b.DomainFacts = []DomainFact{f}
		b.ID = b.ComputeID()
		if err := a.Validate(); (err == nil) != (version == "12" || (version == "13" || version == "14")) {
			t.Fatalf("worker %s: %v", version, err)
		}
	}
}
