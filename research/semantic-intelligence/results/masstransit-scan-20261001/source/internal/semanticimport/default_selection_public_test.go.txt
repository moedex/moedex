package semanticimport_test

import (
	"bytes"
	"context"
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"reflect"
	"strings"
	"testing"

	"moedex/internal/mcp"
	"moedex/internal/semantic"
	"moedex/internal/semanticimport"
	"moedex/internal/semanticindex"
)

func TestPublicDefaultSelection(t *testing.T) {
	stream, root := os.Getenv("MOEDEX_DEFAULT_STREAM"), os.Getenv("MOEDEX_DEFAULT_ROOT")
	if stream == "" || root == "" {
		t.Skip("set MOEDEX_DEFAULT_STREAM/ROOT for native fixture")
	}
	raw, err := os.ReadFile(stream)
	if err != nil {
		t.Fatal(err)
	}
	a, err := semanticimport.Import(context.Background(), bytes.NewReader(raw), semanticimport.Options{Repo: "fixture", Root: root})
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(t.TempDir(), "artifact")
	if err := semantic.Write(path, a); err != nil {
		t.Fatal(err)
	}
	a, err = semantic.Read(path, semantic.Limits{})
	if err != nil {
		t.Fatal(err)
	}
	artifact, _ := os.ReadFile(path)
	sha := fmt.Sprintf("%x", sha256.Sum256(artifact))
	provenance := semanticindex.Provenance{ArtifactSHA256: sha, CorpusFingerprint: strings.Repeat("b", 64)}
	ip := filepath.Join(t.TempDir(), "index")
	if err := semanticindex.Build(ip, a, provenance); err != nil {
		t.Fatal(err)
	}
	x, err := semanticindex.Open(ip, provenance, semanticindex.Limits{})
	if err != nil {
		t.Fatal(err)
	}
	defer x.Close()
	p := &domainGoldProvider{index: x, artifact: sha}
	tools := map[string]mcp.ToolHandler{}
	for _, tool := range mcp.CompilerTools(p) {
		tools[tool.Name()] = tool
	}
	call := func(name string, args any) map[string]any {
		t.Helper()
		raw, _ := json.Marshal(args)
		out, err := tools[name].Call(context.Background(), raw)
		if err != nil {
			t.Fatal(err)
		}
		if out["isError"] == true {
			t.Fatal(out)
		}
		wire, _ := json.Marshal(out["structuredContent"])
		var result map[string]any
		if json.Unmarshal(wire, &result) != nil {
			t.Fatal("bad response")
		}
		return result
	}

	declarations := call("compiler_symbols", map[string]any{"repo": "fixture", "path": "App/Source.cs"})
	if declarations["status"] != "ok" || declarations["truncated"] != false {
		t.Fatal(declarations)
	}
	selected, forwarded := 0, 0
	for _, row := range declarations["results"].([]any) {
		raw, _ := json.Marshal(row)
		var b mcp.CompilerBinding
		if err := json.Unmarshal(raw, &b); err != nil {
			t.Fatal(err)
		}
		for _, f := range b.ImplementationFacts {
			if f.Kind != "interface_default_selection" {
				continue
			}
			selected++
			if b.Symbol.ID != f.ImplementingTypeSymbolID || (b.ExtractorVersion != "8" && b.ExtractorVersion != "9" && (b.ExtractorVersion != "10" && (b.ExtractorVersion != "11" && (b.ExtractorVersion != "12" && (b.ExtractorVersion != "13" && (b.ExtractorVersion != "14" && (b.ExtractorVersion != "15" && (b.ExtractorVersion != "16" && b.ExtractorVersion != "17")))))))) {
				t.Fatal(b)
			}
			sy := map[string]mcp.CompilerSymbol{}
			for _, symbol := range b.ImplementationSymbols {
				sy[symbol.ID] = symbol
			}
			if len(sy) != len(f.SymbolIDs()) || sy[f.SelectedDefaultSymbolID].DescriptorKind != "constructed_interface_method_v1" {
				t.Fatal(sy)
			}
			reverse := call("compiler_implementations", map[string]any{"symbol_id": f.InterfaceSymbolID})
			if reverse["status"] != "context_required" || len(reverse["contexts"].([]any)) != 2 {
				t.Fatal(reverse)
			}
			matches := call("compiler_implementations", map[string]any{"symbol_id": f.InterfaceSymbolID, "context_ids": []string{b.ContextID}})
			if matches["status"] != "ok" || matches["truncated"] != false {
				t.Fatal(matches)
			}
			found := false
			for _, m := range matches["matches"].([]any) {
				raw, _ := json.Marshal(m)
				var match mcp.CompilerImplementationMatch
				json.Unmarshal(raw, &match)
				if match.Source.OccurrenceID == b.OccurrenceID && reflect.DeepEqual(match.Source.ImplementationFacts[match.ImplementationFactIndex], f) {
					found = true
				}
			}
			if !found {
				t.Fatal("class selection absent from reverse query")
			}
			definitions := call("compiler_definitions", map[string]any{"symbol_id": f.DefaultTemplateSymbolID})
			if definitions["status"] != "ok" || len(definitions["results"].([]any)) != 1 {
				t.Fatal(definitions)
			}
			definition := definitions["results"].([]any)[0].(map[string]any)
			if definition["path"] != "Contracts/Source.cs" || definition["raw_sha256"] == "" || definition["context_id"] == b.ContextID {
				t.Fatal(definition)
			}
			if v := f.Forwarding; v != nil {
				forwarded++
				callsite := call("compiler_binding_at", map[string]any{"repo": "fixture", "path": v.CallPath, "byte_offset": v.CallOffset, "context_id": v.CallContextID, "raw_sha256": v.CallSHA256})
				if callsite["status"] != "ok" || len(callsite["results"].([]any)) != 1 {
					t.Fatal(callsite)
				}
				witness := callsite["results"].([]any)[0].(map[string]any)
				if witness["occurrence_id"] != v.CallOccurrenceID || witness["enclosing_symbol_id"] != f.DefaultTemplateSymbolID {
					t.Fatal(witness)
				}
				target := call("compiler_definitions", map[string]any{"symbol_id": v.ImplementationSymbolID, "context_id": b.ContextID})
				if target["status"] != "ok" || len(target["results"].([]any)) != 1 {
					t.Fatal(target)
				}
			}
			// The closed identity is not silently collapsed onto the open declaration.
			closed := call("compiler_definitions", map[string]any{"symbol_id": f.SelectedDefaultSymbolID})
			if len(closed["results"].([]any)) != 0 {
				t.Fatal(closed)
			}
		}
	}
	if (a.Contexts[0].ExtractorVersion == "9" || (a.Contexts[0].ExtractorVersion == "10" || (a.Contexts[0].ExtractorVersion == "11" || (a.Contexts[0].ExtractorVersion == "12" || (a.Contexts[0].ExtractorVersion == "13" || (a.Contexts[0].ExtractorVersion == "14" || (a.Contexts[0].ExtractorVersion == "15" || (a.Contexts[0].ExtractorVersion == "16" || a.Contexts[0].ExtractorVersion == "17")))))))) && forwarded != 3 {
		t.Fatalf("forwarded=%d", forwarded)
	}
	if selected != 6 {
		t.Fatalf("selected=%d", selected)
	}
	if p.acquired != p.released {
		t.Fatal("lease leak")
	}
}
