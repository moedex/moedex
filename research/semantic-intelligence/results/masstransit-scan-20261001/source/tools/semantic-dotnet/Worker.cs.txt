static partial class Worker
{
    const string Schema = "moedex.semantic-worker.v1";
    const string Extractor = "msbuild-roslyn";
    const string ExtractorVersion = "17";
    static readonly UTF8Encoding Utf8 = new(false, true);
    static readonly JsonSerializerOptions Json = new() { PropertyNamingPolicy = JsonNamingPolicy.SnakeCaseLower };
    static string Root = "", Repo = "", Sdk = "";
    static string Compiler => typeof(CSharpCompilation).Assembly.GetName().Version!.ToString();
    static readonly List<State> States = [];
    static readonly Dictionary<string, object> FileInputs = new(StringComparer.Ordinal);
    static readonly Dictionary<string, string> ObservedFiles = new(StringComparer.Ordinal);

    public static async Task<int> Run(Dictionary<string, string> args)
    {
        Root = Path.GetFullPath(args["--root"]); Repo = args["--repo"]; Sdk = Path.GetFullPath(args["--sdk-path"]);
        if (Repo.Contains('\\') || Repo.Any(char.IsControl) || Repo.Split('/').Any(part => part is "" or "." or "..")) throw new ArgumentException("repo must be a clean relative namespace");
        var path = Path.GetFullPath(args["--project"], Root);
        RequireInside(path);
        RequireNoEscapingLink(path);
        var properties = new Dictionary<string, string> { ["Configuration"] = args.GetValueOrDefault("--configuration", "Debug"), ["TargetFramework"] = args["--framework"] };
        using var workspace = MSBuildWorkspace.Create(properties);
        workspace.LoadMetadataForReferencedProjects = false;
        workspace.SkipUnrecognizedProjects = false;
        await workspace.OpenProjectAsync(path);
        foreach (var project in workspace.CurrentSolution.Projects.OrderBy(p => p.FilePath, StringComparer.Ordinal))
        {
            RequireInside(project.FilePath!);
            var state = new State(project);
            States.Add(state);
            if (project.Language != LanguageNames.CSharp) { state.Issues.Add("unsupported_language"); continue; }
            state.Compilation = await project.GetCompilationAsync();
            if (state.Compilation is null) { state.Issues.Add("compilation_unavailable"); continue; }
            state.Diagnostics = state.Compilation.GetDiagnostics().ToArray();
            foreach (var document in await project.GetSourceGeneratedDocumentsAsync())
                if (await document.GetSyntaxTreeAsync() is { } generatedTree) state.GeneratedTrees.Add(generatedTree);
            VerifyGenerators(state);
            foreach (var tree in state.Compilation.SyntaxTrees)
                state.Sources.Add(ReadSource(state, tree));
            state.Sources.Sort((a, b) => StringComparer.Ordinal.Compare(a.Path, b.Path));
            if (state.Sources.Select(s => s.Path).Distinct(StringComparer.Ordinal).Count() != state.Sources.Count) throw new InvalidDataException("duplicate source paths");
        }
        var workspaceIssues = workspace.Diagnostics.OrderBy(d => d.Message, StringComparer.Ordinal).ToArray();
        // A failed design-time load is not rescued by a usable-looking compilation.
        foreach (var state in States)
        {
            if (workspaceIssues.Length != 0) state.Issues.Add("workspace_diagnostics");
            state.Capture = Capture(state, properties);
        }
        // Include referenced contexts transitively, and propagate failed capture/compilation.
        foreach (var state in States) FinalizeContext(state, []);
        int declarations = 0, references = 0;
        foreach (var state in States)
        {
            var counts = EmitProject(state, workspaceIssues);
            declarations += counts.Item1; references += counts.Item2;
        }
        foreach (var observed in ObservedFiles)
            if (!File.Exists(observed.Key) || Hash(File.ReadAllBytes(observed.Key)) != observed.Value) throw new InvalidDataException("captured input changed during extraction");
        var status = States.All(s => s.Complete) ? "complete" : "incomplete";
        Emit(new { Schema, RecordType = "stream_summary", Projects = States.Count, CompilationStatus = status, Declarations = declarations, References = references });
        return status == "complete" ? 0 : 2;
    }

    // Workspaces does not expose generator diagnostics through GetCompilationAsync.
    // Verify with the public driver and the workspace's actual analyzer inputs;
    // disagreement is incomplete, never silently accepted as the same build.
    static void VerifyGenerators(State state)
    {
        var generators = state.Project.AnalyzerReferences.SelectMany(a => a.GetGenerators(LanguageNames.CSharp)).ToArray();
        if (generators.Length == 0) return;
        if (state.Compilation is not CSharpCompilation compilation || state.Project.ParseOptions is not CSharpParseOptions parse)
        { state.Issues.Add("unsupported_generator_context"); return; }
        var input = compilation.RemoveSyntaxTrees(state.GeneratedTrees);
        GeneratorDriver driver = CSharpGeneratorDriver.Create(generators,
            additionalTexts: state.Project.AnalyzerOptions.AdditionalFiles,
            parseOptions: parse,
            optionsProvider: state.Project.AnalyzerOptions.AnalyzerConfigOptionsProvider);
        driver = driver.RunGenerators(input);
        var result = driver.GetRunResult();
        state.Diagnostics = state.Diagnostics.Concat(result.Diagnostics).ToArray();
        if (result.Results.Any(r => r.Exception is not null)) state.Issues.Add("generator_failure");
        string[] Manifest(IEnumerable<SyntaxTree> trees) => trees.Select(t => t.FilePath + "\n" + Hash(Utf8.GetBytes(t.GetText().ToString()))).Order(StringComparer.Ordinal).ToArray();
        state.GeneratorVerification = new { Driver = Manifest(result.GeneratedTrees), Workspace = Manifest(state.GeneratedTrees) };
        // Workspace prefixes generated hint paths with its evaluated obj directory;
        // the public driver uses assembly/generator/hint paths. Require the full
        // driver suffix (not merely basename) and identical text one-to-one.
        var remaining = state.GeneratedTrees.ToList();
        bool mismatch = false;
        foreach (var generated in result.GeneratedTrees)
        {
            var matches = remaining.Where(t => (t.FilePath == generated.FilePath ||
                Path.IsPathRooted(t.FilePath) && t.FilePath.Replace('\\', '/').EndsWith("/" + generated.FilePath.Replace('\\', '/'), StringComparison.Ordinal)) &&
                t.GetText().ContentEquals(generated.GetText())).ToArray();
            if (matches.Length != 1) mismatch = true;
            else remaining.Remove(matches[0]);
        }
        if (mismatch || remaining.Count != 0) state.Issues.Add("generator_workspace_disagreement");
    }

    static Source ReadSource(State state, SyntaxTree tree)
    {
        var text = tree.GetText().ToString();
        var physical = tree.FilePath;
        bool compilerGenerated = state.GeneratedTrees.Contains(tree);
        if (!compilerGenerated)
        {
            if (string.IsNullOrEmpty(physical) || !Path.IsPathRooted(physical) || !File.Exists(physical)) throw new InvalidDataException("missing physical compiler source");
            RequireInside(physical);
            RequireNoEscapingLink(physical);
        }
        bool onDisk = !string.IsNullOrEmpty(physical) && Path.IsPathRooted(physical) && Inside(physical) && File.Exists(physical);
        byte[] bytes; int bom = 0; string path;
        bool generated = !onDisk || physical.Replace('\\', '/').Contains("/obj/", StringComparison.Ordinal) || physical.EndsWith(".g.cs", StringComparison.OrdinalIgnoreCase) || physical.EndsWith(".generated.cs", StringComparison.OrdinalIgnoreCase);
        if (onDisk)
        {
            bytes = File.ReadAllBytes(physical);
            bom = bytes.AsSpan().StartsWith(new byte[] { 0xef, 0xbb, 0xbf }) ? 3 : 0;
            if (Utf8.GetString(bytes.AsSpan(bom)) != text) throw new InvalidDataException("source bytes differ from compiler text: " + Relative(physical));
            path = Relative(physical);
            Observe(physical, bytes);
        }
        else
        {
            bytes = Utf8.GetBytes(text);
            path = ".generated/" + Hash(Utf8.GetBytes(Relative(state.Project.FilePath!) + "\n" + physical))[..24] + "/" + Path.GetFileName(physical);
            if (path.EndsWith('/')) path += "generated.cs";
        }
        return new Source(path, bytes, text, bom, tree, generated);
    }

    static Dictionary<string, object?> Capture(State s, Dictionary<string, string> properties)
    {
        var capture = new Dictionary<string, object?>
        {
            ["project"] = Relative(s.Project.FilePath!), ["repo"] = Repo,
            ["sources"] = s.Sources.Select(x => x.Manifest(false)).ToArray(),
            ["extractor"] = Extractor, ["extractor_version"] = ExtractorVersion, ["compiler_version"] = Compiler,
            ["global_properties"] = properties,
            ["project_file"] = FileInput(s.Project.FilePath!),
            ["generator_verification"] = s.GeneratorVerification,
            ["compiler_assembly"] = FileInput(typeof(CSharpCompilation).Assembly.Location),
            ["workspace_assembly"] = FileInput(typeof(MSBuildWorkspace).Assembly.Location),
            ["extractor_assembly"] = FileInput(typeof(Worker).Assembly.Location),
        };
        using var collection = new ProjectCollection(properties);
        var evaluated = collection.LoadProject(s.Project.FilePath!);
        var names = new[] { "Configuration", "Platform", "TargetFramework", "TargetFrameworks", "RuntimeIdentifier", "DefineConstants", "LangVersion", "Nullable", "AllowUnsafeBlocks", "CheckForOverflowUnderflow", "Optimize", "WarningLevel", "NoWarn", "WarningsAsErrors", "TreatWarningsAsErrors", "AssemblyName", "AssemblyVersion", "Version", "OutputType", "Deterministic", "SignAssembly", "PublicSign", "ImplicitUsings", "MSBuildVersion", "NETCoreSdkVersion" };
        capture["evaluated_properties"] = names.ToDictionary(n => n, n => evaluated.GetPropertyValue(n));
        capture["imports"] = evaluated.Imports.Select(i => i.ImportedProject.FullPath).Distinct(StringComparer.Ordinal).Order(StringComparer.Ordinal).Select(FileInput).ToArray();
        capture["packages"] = evaluated.GetItems("PackageReference").OrderBy(i => i.EvaluatedInclude, StringComparer.Ordinal).Select(i => new { Name = i.EvaluatedInclude, Metadata = i.Metadata.OrderBy(m => m.Name, StringComparer.Ordinal).ToDictionary(m => m.Name, m => m.EvaluatedValue) }).ToArray();
        var assets = evaluated.GetPropertyValue("ProjectAssetsFile");
        if (!string.IsNullOrEmpty(assets))
        {
            assets = Path.GetFullPath(assets, Path.GetDirectoryName(s.Project.FilePath!)!);
            if (File.Exists(assets)) capture["package_assets"] = FileInput(assets);
            else s.Issues.Add("missing_package_assets");
        }
        capture["metadata_references"] = s.Project.MetadataReferences.OrderBy(r => r.Display, StringComparer.Ordinal).Select(r => new { Input = r is PortableExecutableReference pe && pe.FilePath is not null ? FileInput(pe.FilePath) : throw new InvalidDataException("metadata reference has no file identity"), Aliases = r.Properties.Aliases.Order(StringComparer.Ordinal).ToArray(), r.Properties.EmbedInteropTypes, Kind = r.Properties.Kind.ToString() }).ToArray();
        capture["analyzers"] = s.Project.AnalyzerReferences.OrderBy(r => r.FullPath, StringComparer.Ordinal).Select(r => FileInput(r.FullPath ?? throw new InvalidDataException("analyzer has no file path"))).ToArray();
        capture["additional_files"] = s.Project.AdditionalDocuments.OrderBy(d => d.FilePath, StringComparer.Ordinal).Select(d => FileInput(d.FilePath!)).ToArray();
        capture["analyzer_configs"] = s.Project.AnalyzerConfigDocuments.OrderBy(d => d.FilePath, StringComparer.Ordinal).Select(d => FileInput(d.FilePath!)).ToArray();
        if (s.Project.ParseOptions is CSharpParseOptions parse)
            capture["parse_options"] = new { LanguageVersion = parse.LanguageVersion.ToString(), DocumentationMode = parse.DocumentationMode.ToString(), Kind = parse.Kind.ToString(), Defines = parse.PreprocessorSymbolNames.Order(StringComparer.Ordinal).ToArray(), Features = parse.Features.OrderBy(p => p.Key, StringComparer.Ordinal).ToDictionary(p => p.Key, p => p.Value) };
        if (s.Project.CompilationOptions is CSharpCompilationOptions options)
            capture["compilation_options"] = new { OutputKind = options.OutputKind.ToString(), options.ModuleName, options.MainTypeName, options.ScriptClassName, OptimizationLevel = options.OptimizationLevel.ToString(), options.CheckOverflow, options.AllowUnsafe, Platform = options.Platform.ToString(), options.WarningLevel, GeneralDiagnosticOption = options.GeneralDiagnosticOption.ToString(), SpecificDiagnosticOptions = options.SpecificDiagnosticOptions.OrderBy(p => p.Key, StringComparer.Ordinal).ToDictionary(p => p.Key, p => p.Value.ToString()), NullableContextOptions = options.NullableContextOptions.ToString(), options.Deterministic, options.ConcurrentBuild, options.CryptoKeyContainer, CryptoKeyFile = string.IsNullOrEmpty(options.CryptoKeyFile) ? null : FileInput(Path.GetFullPath(options.CryptoKeyFile, Path.GetDirectoryName(s.Project.FilePath!)!)), options.DelaySign, options.PublicSign, Usings = options.Usings.Order(StringComparer.Ordinal).ToArray() };
        return capture;
    }

    static void FinalizeContext(State s, HashSet<State> active)
    {
        if (s.Context.Length != 0) return;
        if (!active.Add(s)) throw new InvalidDataException("cyclic project references");
        foreach (var reference in s.Project.ProjectReferences)
        {
            var target = States.SingleOrDefault(t => t.Project.Id == reference.ProjectId);
            if (target is null) { s.Issues.Add("missing_project_reference"); continue; }
            FinalizeContext(target, active);
            s.Dependencies.Add(target);
            if (!target.Complete) s.Issues.Add("incomplete_project_reference");
        }
        active.Remove(s);
        s.Capture["project_references"] = s.Project.ProjectReferences.OrderBy(r => States.Single(t => t.Project.Id == r.ProjectId).Project.FilePath, StringComparer.Ordinal).Select(r => { var d = States.Single(t => t.Project.Id == r.ProjectId); return new { Project = Relative(d.Project.FilePath!), BuildContext = d.Context, Aliases = r.Aliases.Order(StringComparer.Ordinal).ToArray(), r.EmbedInteropTypes }; }).ToArray();
        s.CaptureJSON = JsonSerializer.Serialize(s.Capture, Json);
        s.Context = Hash(Utf8.GetBytes(s.CaptureJSON));
    }

    static (int, int) EmitProject(State s, WorkspaceDiagnostic[] workspaceIssues)
    {
        string status = s.Complete ? "complete" : "incomplete";
        Emit(new { Schema, RecordType = "project", Project = Relative(s.Project.FilePath!), Repo, BuildContext = s.Context, CompilationStatus = status, Extractor, ExtractorVersion, CompilerVersion = Compiler, CaptureJSON = s.CaptureJSON, Sources = s.Sources.Select(x => x.Manifest(true)).ToArray(), ProjectReferences = s.Dependencies.OrderBy(d => d.Project.FilePath, StringComparer.Ordinal).Select(d => new { Project = Relative(d.Project.FilePath!), BuildContext = d.Context }).ToArray(), Issues = s.Issues.Order(StringComparer.Ordinal).ToArray() });
        int declarations = 0, references = 0;
        if (s.Compilation is not null)
        foreach (var source in s.Sources)
        {
            var model = s.Compilation.GetSemanticModel(source.Tree);
            foreach (var node in source.Tree.GetRoot().DescendantNodesAndSelf())
            {
                var token = DeclarationToken(node);
                if (token.RawKind != 0 && model.GetDeclaredSymbol(node) is { } declared)
                { Occurrence(s, source, "declaration", token.Span, declared, [], "resolved", "declaration", null, implementationFacts: ImplementationFacts(s, declared)); declarations++; }
                // Primary constructors share the type identifier span but have a
                // distinct symbol and occurrence kind. Without this declaration,
                // resolved constructor calls cannot navigate to their source.
                if (node is TypeDeclarationSyntax { ParameterList: not null } type && model.GetDeclaredSymbol(type) is INamedTypeSymbol namedType)
                {
                    foreach (var constructor in namedType.InstanceConstructors.Where(c => c.DeclaringSyntaxReferences.Any(r => r.SyntaxTree == source.Tree && r.Span == type.Span)))
                    { Occurrence(s, source, "declaration", type.Identifier.Span, constructor, [], "resolved", "constructor_declaration", null); declarations++; }
                }
                if (node is SimpleNameSyntax name)
                {
                    SyntaxNode bind = name;
                    if (name.Parent is MemberAccessExpressionSyntax member && member.Name == name && member.Parent is InvocationExpressionSyntax inv && inv.Expression == member) bind = inv;
                    else if (name.Parent is InvocationExpressionSyntax direct && direct.Expression == name) bind = direct;
                    else if (name.Parent is MemberBindingExpressionSyntax binding && binding.Parent is InvocationExpressionSyntax conditional) bind = conditional;
                    var info = model.GetSymbolInfo(bind);
                    var symbol = info.Symbol is IErrorTypeSymbol ? null : info.Symbol;
                    var statusName = symbol is not null ? "resolved" : info.CandidateSymbols.Length > 1 ? "ambiguous" : "unresolved";
                    var domainFacts = DomainFacts(s, model, name, bind, symbol, out var domainOwner);
                    Occurrence(s, source, "reference", name.Identifier.Span, symbol, info.CandidateSymbols, statusName, bind is InvocationExpressionSyntax ? "invocation" : "name", domainOwner ?? model.GetEnclosingSymbol(name.SpanStart), domainFacts); references++;
                }
                if (node is BaseObjectCreationExpressionSyntax creation)
                {
                    var info = model.GetSymbolInfo(creation);
                    var span = creation is ObjectCreationExpressionSyntax explicitCreation ? TypeToken(explicitCreation.Type).Span : creation.NewKeyword.Span;
                    Occurrence(s, source, "reference", span, info.Symbol, info.CandidateSymbols, info.Symbol is not null ? "resolved" : info.CandidateSymbols.Length > 1 ? "ambiguous" : "unresolved", "constructor", model.GetEnclosingSymbol(creation.SpanStart)); references++;
                }
            }
        }
        foreach (var d in s.Diagnostics.OrderBy(d => d.Location.SourceTree?.FilePath, StringComparer.Ordinal).ThenBy(d => d.Location.SourceSpan.Start).ThenBy(d => d.Id, StringComparer.Ordinal))
        {
            var source = s.Sources.FirstOrDefault(x => x.Tree == d.Location.SourceTree);
            Emit(new { Schema, RecordType = "diagnostic", Project = Relative(s.Project.FilePath!), BuildContext = s.Context, Code = d.Id, Severity = d.Severity.ToString().ToLowerInvariant(), Message = d.GetMessage(CultureInfo.InvariantCulture), SourcePath = source?.Path, Span = source is null ? null : Span(source, d.Location.SourceSpan) });
        }
        foreach (var issue in s.Issues.Distinct().Order(StringComparer.Ordinal))
            Emit(new { Schema, RecordType = "diagnostic", Project = Relative(s.Project.FilePath!), BuildContext = s.Context, Code = issue, Severity = "error", Message = issue });
        foreach (var issue in workspaceIssues)
            Emit(new { Schema, RecordType = "diagnostic", Project = Relative(s.Project.FilePath!), BuildContext = s.Context, Code = "workspace", Severity = "error", Message = issue.Message });
        Emit(new { Schema, RecordType = "summary", Project = Relative(s.Project.FilePath!), BuildContext = s.Context, CompilationStatus = status, Declarations = declarations, References = references, Errors = s.Diagnostics.Count(d => d.Severity == DiagnosticSeverity.Error) + s.Issues.Distinct().Count() + workspaceIssues.Length });
        return (declarations, references);
    }

    static void Occurrence(State state, Source source, string kind, TextSpan span, ISymbol? symbol, IEnumerable<ISymbol> candidates, string status, string referenceKind, ISymbol? enclosing, object[]? domainFacts = null, object[]? implementationFacts = null)
    {
        var row = new { Schema, RecordType = kind, Project = Relative(state.Project.FilePath!), BuildContext = state.Context, SourcePath = source.Path, SourceSha256 = source.Sha, Span = Span(source, span), SourceText = source.Text.Substring(span.Start, span.Length), BindingStatus = status, BindingMethod = "roslyn-semantic-model", ReferenceKind = referenceKind, Symbol = Symbol(symbol), Candidates = candidates.Select(Symbol).OrderBy(x => JsonSerializer.Serialize(x, Json), StringComparer.Ordinal).ToArray(), EnclosingSymbol = Symbol(enclosing) };
        if (domainFacts is null && implementationFacts is null) { Emit(row); return; }
        var enriched = JsonSerializer.SerializeToElement(row, Json).EnumerateObject().ToDictionary(p => p.Name, p => (object)p.Value);
        if (domainFacts is not null) enriched["domain_facts"] = domainFacts;
        if (implementationFacts is not null) enriched["implementation_facts"] = implementationFacts;
        Emit(enriched);
    }
    static object? Symbol(ISymbol? input)
    {
        if (input is null) return null;
        if (input is IMethodSymbol closed && ClosedInterfaceMethod(closed)) return ConstructedInterfaceSymbol(closed);
        if (input is IMethodSymbol extended && ExtendedInterfaceSymbol(extended) is { } key) return key;
        var symbol = input is IMethodSymbol { ReducedFrom: { } reduced } ? reduced.OriginalDefinition : input.OriginalDefinition;
        var descriptor = symbol.GetDocumentationCommentId();
        string descriptorKind = descriptor is null ? "location_fallback" : "documentation_comment_id";
        var location = symbol.Locations.FirstOrDefault(l => l.IsInSource);
        var owner = States.FirstOrDefault(s => s.Compilation is not null && SymbolEqualityComparer.Default.Equals(s.Compilation.Assembly, symbol.ContainingAssembly));
        if (owner is null && location is not null && symbol is not INamespaceSymbol)
        {
            var matches = States.Where(s => s.Compilation?.Assembly.Identity.Equals(symbol.ContainingAssembly?.Identity) == true && s.Sources.Any(x => x.Tree.FilePath == location.SourceTree?.FilePath)).ToArray();
            if (matches.Length == 1) owner = matches[0];
            else throw new InvalidDataException("cannot identify source symbol project");
        }
        var source = owner?.Sources.FirstOrDefault(x => x.Tree.FilePath == location?.SourceTree?.FilePath);
        descriptor ??= symbol.ToDisplayString(SymbolDisplayFormat.FullyQualifiedFormat) + "@" + (source is null ? "metadata" : source.Path + ":" + location!.SourceSpan.Start.ToString(CultureInfo.InvariantCulture));
        return new { Descriptor = descriptor, DescriptorKind = descriptorKind, NamespaceKind = owner is null ? "assembly" : "project", Namespace = owner is null ? symbol.ContainingAssembly?.Identity.ToString() ?? "<global>" : Repo + "/" + Relative(owner.Project.FilePath!), Language = "csharp", DisplayName = symbol.ToDisplayString(SymbolDisplayFormat.CSharpErrorMessageFormat), Kind = symbol.Kind.ToString() };
    }
    static object Span(Source source, TextSpan span) => new { ByteOffset = source.Bom + Utf8.GetByteCount(source.Text.AsSpan(0, span.Start)), ByteLength = Utf8.GetByteCount(source.Text.AsSpan(span.Start, span.Length)) };
    static string Hash(byte[] bytes) => Convert.ToHexStringLower(SHA256.HashData(bytes));
    static string Relative(string path) => Path.GetRelativePath(Root, path).Replace('\\', '/');
    static bool Inside(string path) { var relative = Relative(path); return relative != ".." && !relative.StartsWith("../", StringComparison.Ordinal) && !Path.IsPathRooted(relative); }
    static void RequireInside(string path) { if (!Inside(path)) throw new InvalidDataException("project is outside root"); }
    static void RequireNoEscapingLink(string path)
    {
        for (string? current = Path.GetFullPath(path); current is not null && Inside(current); current = Path.GetDirectoryName(current))
        {
            FileSystemInfo info = Directory.Exists(current) ? new DirectoryInfo(current) : new FileInfo(current);
            var target = info.ResolveLinkTarget(true);
            if (target is not null && !Inside(target.FullName)) throw new InvalidDataException("source symlink escapes root");
            if (current == Root) break;
        }
    }
    static object FileInput(string path)
    {
        path = Path.GetFullPath(path);
        if (FileInputs.TryGetValue(path, out var cached)) return cached;
        var bytes = File.ReadAllBytes(path);
        Observe(path, bytes);
        string name = Inside(path) ? Relative(path) : path.StartsWith(Sdk + Path.DirectorySeparatorChar, StringComparison.Ordinal) ? "sdk/" + Path.GetRelativePath(Sdk, path).Replace('\\', '/') : "external/" + Path.GetFileName(path);
        return FileInputs[path] = new { Path = name, Scope = Inside(path) ? "source" : path.StartsWith(Sdk + Path.DirectorySeparatorChar, StringComparison.Ordinal) ? "sdk" : "external", Sha256 = Hash(bytes), ByteSize = bytes.Length };
    }
    static void Observe(string path, byte[] bytes)
    {
        var digest = Hash(bytes);
        if (ObservedFiles.TryGetValue(path, out var before) && before != digest) throw new InvalidDataException("captured input changed during extraction");
        ObservedFiles[path] = digest;
    }
    static void Emit(object row) => Console.WriteLine(JsonSerializer.Serialize(row, Json));
    sealed class State(Microsoft.CodeAnalysis.Project project)
    {
        public Microsoft.CodeAnalysis.Project Project = project;
        public Compilation? Compilation;
        public Diagnostic[] Diagnostics = [];
        public List<Source> Sources = [];
        public HashSet<SyntaxTree> GeneratedTrees = [];
        public List<string> Issues = [];
        public List<State> Dependencies = [];
        public Dictionary<string, object?> Capture = [];
        public string CaptureJSON = "", Context = "";
        public object? GeneratorVerification;
        public bool Complete => Compilation is not null && Issues.Count == 0 && !Diagnostics.Any(d => d.Severity == DiagnosticSeverity.Error);
    }
    sealed class Source(string path, byte[] bytes, string text, int bom, SyntaxTree tree, bool generated)
    {
        public string Path = path, Text = text, Sha = Hash(bytes);
        public byte[] Bytes = bytes;
        public int Bom = bom;
        public SyntaxTree Tree = tree;
        public object Manifest(bool embed) => new { Path, Sha256 = Sha, ByteSize = Bytes.Length, Generated = generated, ContentBase64 = embed && generated ? Convert.ToBase64String(Bytes) : null };
    }
}
