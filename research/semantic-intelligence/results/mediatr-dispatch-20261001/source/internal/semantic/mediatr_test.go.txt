func mediatorResponseKey() SymbolKey {
	arg := SymbolKey{Language: "csharp", NamespaceKind: "project", Namespace: "repo/Contracts.csproj", Descriptor: "T:One.Reply", DescriptorKind: "documentation_comment_id"}
	raw, _ := json.Marshal(struct {
		Definition string      `json:"definition"`
		Arguments  []SymbolKey `json:"arguments"`
	}{"T:System.Collections.Generic.IEnumerable`1", []SymbolKey{arg}})
	return SymbolKey{Language: "csharp", NamespaceKind: "assembly", Namespace: "System.Runtime, Version=8.0.0.0", Descriptor: string(raw), DescriptorKind: "constructed_named_type_v1"}
}
func TestMediatorConstructedResponseIdentity(t *testing.T) {
	k := mediatorResponseKey()
	if err := ValidateConstructedNamedType(k); err != nil {
		t.Fatal(err)
	}
	other := k
	other.Descriptor = strings.ReplaceAll(k.Descriptor, "One.Reply", "Two.Reply")
	if err := ValidateConstructedNamedType(other); err != nil {
		t.Fatal(err)
	}
	if (Symbol{Key: k}).ComputeID() == (Symbol{Key: other}).ComputeID() {
		t.Fatal("qualified arguments collapsed")
	}
	for _, change := range []func(*SymbolKey){
		func(k *SymbolKey) { k.Descriptor += "{}" }, func(k *SymbolKey) { k.Descriptor = strings.ReplaceAll(k.Descriptor, "`1", "`2") }, func(k *SymbolKey) { k.Descriptor = strings.ReplaceAll(k.Descriptor, "`1", "`01") }, func(k *SymbolKey) { k.Descriptor = strings.ReplaceAll(k.Descriptor, "T:One.Reply", "T:One.Reply`1") }, func(k *SymbolKey) { k.Descriptor = strings.ReplaceAll(k.Descriptor, "T:One.Reply", "T:One.Reply[]") }, func(k *SymbolKey) { k.NamespaceKind = "package" }, func(k *SymbolKey) { k.Descriptor = strings.Repeat("x", 32769) }, func(k *SymbolKey) {
			k.Descriptor = strings.ReplaceAll(k.Descriptor, `"definition":`, `"unknown":1,"definition":`)
		},
	} {
		bad := k
		change(&bad)
		if ValidateConstructedNamedType(bad) == nil {
			t.Fatalf("invalid key accepted %+v", bad)
		}
	}
	response := Symbol{Key: k}
	response.ID = response.ComputeID()
	f := DomainFact{Kind: "mediator_send_configuration", Rule: "csharp-mediatr-v1", EvidenceScope: "compile_time"}
	if err := ValidateDomainFactTarget(f, "response", response); err != nil {
		t.Fatal(err)
	}
	if ValidateDomainFactTarget(f, "request", response) == nil || ValidateDomainFactTarget(DomainFact{Kind: "message_response"}, "message", response) == nil {
		t.Fatal("generic response support escaped rule boundary")
	}
}
func TestMediatorFactArtifactAndVersion(t *testing.T) {
	a := domainFixture()
	api := a.Symbols[1]
	api.Key.Namespace = "MediatR, Version=12.0.0.0"
	api.Key.Descriptor = "M:MediatR.ISender.Send``1(MediatR.IRequest{``0},System.Threading.CancellationToken)"
	api.ID = api.ComputeID()
	response := Symbol{Key: mediatorResponseKey()}
	response.ID = response.ComputeID()
	a.Symbols = append(a.Symbols, response)
	f := DomainFact{Kind: "mediator_send_configuration", Rule: "csharp-mediatr-v1", EvidenceScope: "compile_time", Targets: []DomainTarget{{Role: "request", SymbolID: a.Symbols[0].ID}, {Role: "response", SymbolID: response.ID}}}
	if err := ValidateDomainFacts([]DomainFact{f}); err != nil {
		t.Fatal(err)
	}
	if err := ValidateDomainAPI(f, api); err != nil {
		t.Fatal(err)
	}
	for _, change := range []func(*DomainFact){func(f *DomainFact) { f.Rule = "csharp-framework-v6" }, func(f *DomainFact) { f.EvidenceScope = "runtime" }, func(f *DomainFact) { f.Lifetime = "scoped" }, func(f *DomainFact) { f.RoutePattern = "/x" }, func(f *DomainFact) { f.Targets = f.Targets[:1] }, func(f *DomainFact) { f.Targets[0].Role = "handler" }} {
		bad := f
		bad.Targets = append([]DomainTarget(nil), f.Targets...)
		change(&bad)
		if ValidateDomainFacts([]DomainFact{bad}) == nil {
			t.Fatal(bad)
		}
	}
	for _, change := range []func(*Symbol){func(s *Symbol) { s.Key.NamespaceKind = "project" }, func(s *Symbol) { s.Key.Namespace = "Lookalike" }, func(s *Symbol) { s.Key.Descriptor = strings.ReplaceAll(s.Key.Descriptor, "ISender", "Mediator") }, func(s *Symbol) {
		s.Key.Descriptor = "M:MediatR.ISender.Send(System.Object,System.Threading.CancellationToken)"
	}} {
		bad := api
		change(&bad)
		if ValidateDomainAPI(f, bad) == nil {
			t.Fatal(bad)
		}
	}
	for _, version := range []string{"12", "13", "14"} {
		a.Symbols[1] = api
		c := &a.Contexts[0]
		c.ExtractorVersion = version
		c.ID = c.ComputeID()
		o := &a.Occurrences[0]
		o.ContextID = c.ID
		o.ID = o.ComputeID()
		b := &a.Bindings[0]
		b.ExtractorVersion = version
		b.OccurrenceID = o.ID
		b.SymbolID = api.ID
		b.DomainFacts = []DomainFact{f}
		b.ID = b.ComputeID()
		if err := a.Validate(); (err == nil) != (version == "13") {
			t.Fatalf("worker %s: %v", version, err)
		}
	}
}
