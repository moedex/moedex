using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using MediatR;
namespace One { class Reply {} }
namespace Two { class Reply {} }
class Detail : IRequest<One.Reply> { public int Value {get;set;} }
class Orders : IRequest<IEnumerable<One.Reply>> {}
class OtherOrders : IRequest<IEnumerable<Two.Reply>> {}
class DictionaryRequest : IRequest<Dictionary<string,One.Reply>> {}
class Inherited : Detail {}
class Nested : IRequest<List<List<One.Reply>>> {}
class ArrayRequest : IRequest<One.Reply[]> {}
class TupleRequest : IRequest<(int,string)> {}
class GenericRequest<T> : IRequest<One.Reply> {}
class NoResponse : IRequest {}
class BaseReply {}
class DerivedReply : BaseReply {}
class Covariant : IRequest<DerivedReply> {}
class Nine<A,B,C,D,E,F,G,H,I> {}
class TooWide : IRequest<Nine<int,int,int,int,int,int,int,int,int>> {}
class Fake { public Task<T> Send<T>(IRequest<T> request) => Task.FromResult(default(T)); }
class Calls {
 static Detail Factory() => new Detail();
 async Task Run(ISender sender, IMediator mediator, Mediator concrete, bool enabled) {
  await sender./*simple*/Send(new Detail());
  await mediator./*collection*/Send(new Orders());
  await sender./*other-collection*/Send(new OtherOrders());
  await sender./*dictionary*/Send(new DictionaryRequest());
  await sender./*named*/Send(cancellationToken:default, request:new Detail { Value=1 });
  await sender./*inherited*/Send(new Inherited());
  if(enabled) await sender./*conditional*/Send(new Detail());
  Detail variable=new Detail();await sender./*variable*/Send(variable);
  IRequest<One.Reply> erased=new Detail();await sender./*erased*/Send(erased);
  await sender./*factory*/Send(Factory());
  await sender./*cast*/Send((IRequest<One.Reply>)new Detail());
  await sender./*object*/Send((object)new Detail());
  await sender./*void*/Send(new NoResponse());
  await sender./*nested*/Send(new Nested());
  await sender./*array*/Send(new ArrayRequest());
  await sender./*tuple*/Send(new TupleRequest());
  await sender./*generic-request*/Send(new GenericRequest<int>());
  await sender./*covariant*/Send<BaseReply>(new Covariant());
  await sender./*wide*/Send(new TooWide());
  await new Fake()./*lookalike*/Send(new Detail());
  await concrete./*concrete-api*/Send(new Detail());
  await sender./*conditional-value*/Send(enabled ? new Detail() : variable);
 }
 async Task Open<T>(ISender sender) {await sender./*open*/Send(new GenericRequest<T>());}
}
