using System;
using System.Threading.Tasks;
using MassTransit;
namespace One { class Request {} }
namespace Two { class Request {} }
class Accepted {}
class Rejected {}
class Third {}
class Generic<T> {}
interface IChild : IRequestClient<One.Request> {}
class Fake { public Task GetResponse<A,B>(object value) => Task.CompletedTask; }
class Calls {
 async Task Run(IRequestClient<One.Request> one, IRequestClient<Two.Request> two, IChild child, IRequestClient<Generic<int>> generic) {
  await one./*initializer*/GetResponse<Accepted,Rejected>(new {});
  await one./*typed*/GetResponse<Accepted,Rejected>(new One.Request());
  await two./*other-request*/GetResponse<Accepted,Rejected>(new {});
  await one./*reversed*/GetResponse<Rejected,Accepted>(new {});
  await child./*inherited*/GetResponse<Accepted,Rejected>(new {});
  await one./*repeated*/GetResponse<Accepted,Accepted>(new {});
  await one./*callback*/GetResponse<Accepted,Rejected>(new {}, x => {});
  await one./*single*/GetResponse<Accepted>(new {});
  await one./*three*/GetResponse<Accepted,Rejected,Third>(new {});
  await one./*generic-response*/GetResponse<Generic<int>,Rejected>(new {});
  await generic./*generic-request*/GetResponse<Accepted,Rejected>(new {});
  await new Fake()./*lookalike*/GetResponse<Accepted,Rejected>(new {});
 }
 async Task Open<T>(IRequestClient<T> client, IRequestClient<One.Request> one) where T:class {
  await client./*open-request*/GetResponse<Accepted,Rejected>(new {});
  await one./*open-response*/GetResponse<T,Rejected>(new {});
 }
}
