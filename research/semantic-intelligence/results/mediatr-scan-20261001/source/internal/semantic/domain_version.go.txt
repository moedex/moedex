package semantic

import "fmt"

// ValidateDomainExtractor binds each rule generation to a worker version that
// implements it. Callers separately validate fact shape, API, and context status.
func ValidateDomainExtractor(f DomainFact, version string) error {
	if f.Rule == "csharp-framework-v1" && (version == "2" || version == "3" || version == "4" || version == "5" || version == "6" || version == "7" || version == "8" || version == "9" || (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))))) {
		return nil
	}
	if f.Rule == "csharp-framework-v2" && (version == "3" || version == "4" || version == "5" || version == "6" || version == "7" || version == "8" || version == "9" || (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))))) {
		return nil
	}
	if f.Rule == "csharp-framework-v3" && (version == "4" || version == "5" || version == "6" || version == "7" || version == "8" || version == "9" || (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))))) {
		return nil
	}
	if f.Rule == "csharp-framework-v4" && (version == "5" || version == "6" || version == "7" || version == "8" || version == "9" || (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))))) {
		return nil
	}
	if f.Rule == "csharp-keyed-di-v1" && (version == "7" || version == "8" || version == "9" || (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))))) {
		return nil
	}
	if f.Rule == "csharp-framework-v5" && (version == "10" || (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16")))))) {
		return nil
	}
	if f.Rule == "csharp-framework-v6" && (version == "11" || (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16"))))) {
		return nil
	}
	if f.Rule == "csharp-endpoint-v1" && (version == "12" || (version == "13" || (version == "14" || (version == "15" || version == "16")))) {
		return nil
	}
	if f.Rule == "csharp-mediatr-v1" && (version == "13" || (version == "14" || (version == "15" || version == "16"))) {
		return nil
	}
	if f.Rule == "csharp-open-di-v1" && (version == "15" || version == "16") {
		return nil
	}
	if f.Rule == "csharp-mediatr-scan-v1" && version == "16" {
		return nil
	}
	return fmt.Errorf("semantic: domain rule requires a supported compiler extractor version")
}
