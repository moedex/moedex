#!/usr/bin/env python3
"""Two-response request configuration against pinned real framework metadata."""
import argparse
import json
import os
from pathlib import Path
import subprocess

p = argparse.ArgumentParser(description=__doc__)
for key in ('dotnet', 'sdk', 'worker', 'packages', 'output'):
    p.add_argument('--' + key, required=True, type=Path)
p.add_argument('--expected-worker-version', choices=['11','12','13','14','15','16'], default='16')
a = p.parse_args()
a.output.mkdir(parents=True, exist_ok=False)
source = 'using System;\nusing System.Threading.Tasks;\nusing MassTransit;\nnamespace One { class Request {} }\nnamespace Two { class Request {} }\nclass Accepted {}\nclass Rejected {}\nclass Third {}\nclass Generic<T> {}\ninterface IChild : IRequestClient<One.Request> {}\nclass Fake { public Task GetResponse<A,B>(object value) => Task.CompletedTask; }\nclass Calls {\n async Task Run(IRequestClient<One.Request> one, IRequestClient<Two.Request> two, IChild child, IRequestClient<Generic<int>> generic) {\n  await one./*initializer*/GetResponse<Accepted,Rejected>(new {});\n  await one./*typed*/GetResponse<Accepted,Rejected>(new One.Request());\n  await two./*other-request*/GetResponse<Accepted,Rejected>(new {});\n  await one./*reversed*/GetResponse<Rejected,Accepted>(new {});\n  await child./*inherited*/GetResponse<Accepted,Rejected>(new {});\n  await one./*repeated*/GetResponse<Accepted,Accepted>(new {});\n  await one./*callback*/GetResponse<Accepted,Rejected>(new {}, x => {});\n  await one./*single*/GetResponse<Accepted>(new {});\n  await one./*three*/GetResponse<Accepted,Rejected,Third>(new {});\n  await one./*generic-response*/GetResponse<Generic<int>,Rejected>(new {});\n  await generic./*generic-request*/GetResponse<Accepted,Rejected>(new {});\n  await new Fake()./*lookalike*/GetResponse<Accepted,Rejected>(new {});\n }\n async Task Open<T>(IRequestClient<T> client, IRequestClient<One.Request> one) where T:class {\n  await client./*open-request*/GetResponse<Accepted,Rejected>(new {});\n  await one./*open-response*/GetResponse<T,Rejected>(new {});\n }\n}\n'
(a.output/'Fixture.cs').write_text(source)
(a.output/'global.json').write_text(json.dumps({'sdk': {'version': a.sdk.name, 'rollForward': 'disable'}}))
(a.output/'Fixture.csproj').write_text('<Project Sdk="Microsoft.NET.Sdk"><PropertyGroup><TargetFramework>net8.0</TargetFramework></PropertyGroup><ItemGroup><FrameworkReference Include="Microsoft.AspNetCore.App"/><PackageReference Include="MassTransit.Abstractions" Version="8.2.1"/></ItemGroup></Project>')
(a.output/'NuGet.Config').write_text('<configuration><packageSources><clear /></packageSources></configuration>')
env = dict(os.environ, NUGET_PACKAGES=str(a.packages.resolve()), DOTNET_NOLOGO='1', DOTNET_CLI_TELEMETRY_OPTOUT='1', DOTNET_PROCESSOR_COUNT='2')
subprocess.run([str(a.dotnet),str(a.sdk/'MSBuild.dll'),'-target:Restore',str(a.output/'Fixture.csproj'),'-p:NuGetAudit=false'], env=env, cwd=a.output, check=True, timeout=120)
command = [str(a.dotnet),str(a.worker),'--repo','fixture','--root',str(a.output),'--project','Fixture.csproj','--framework','net8.0','--sdk-path',str(a.sdk)]
result = subprocess.run(command,env=env,capture_output=True,timeout=120)
(a.output/'capture.jsonl').write_bytes(result.stdout)
(a.output/'capture.stderr').write_bytes(result.stderr)
assert result.returncode == 0, (result.returncode,result.stderr)
rows = list(map(json.loads,result.stdout.splitlines()))
assert all(r['extractor_version']==a.expected_worker_version and r['compilation_status']=='complete' for r in rows if r['record_type']=='project')
def at(label):
    offset=source.index('/*'+label+'*/')+len(label)+4
    hits=[r for r in rows if r.get('source_path')=='Fixture.cs' and r.get('span',{}).get('byte_offset')==offset and r['record_type']=='reference']
    assert len(hits)==1,(label,hits)
    return hits[0]
expected={
 'initializer':['One.Request','Accepted','Rejected'],
 'typed':['One.Request','Accepted','Rejected'],
 'other-request':['Two.Request','Accepted','Rejected'],
 'reversed':['One.Request','Rejected','Accepted'],
 'inherited':['One.Request','Accepted','Rejected'],
 'repeated':['One.Request','Accepted','Accepted'],
}
for label,targets in expected.items():
    facts=at(label).get('domain_facts',[])
    assert len(facts)==1,(label,facts)
    f=facts[0]
    assert f['kind']=='message_request_configuration' and f['rule']=='csharp-framework-v6' and f['evidence_scope']=='compile_time',f
    assert [(t['role'],t['symbol']['descriptor']) for t in f['targets']]==list(zip(['request','response_1','response_2'],['T:'+t for t in targets])),f
negatives=['callback','single','three','generic-response','generic-request','lookalike','open-request','open-response']
for label in negatives: assert not at(label).get('domain_facts'),(label,at(label))
(a.output/'expected.json').write_text(json.dumps({'positive':expected,'negative':negatives},indent=2)+'\n')
(a.output/'Fixture.cs').write_text(source+'\nclass Broken { MissingType value; }')
broken=subprocess.run(command,env=env,capture_output=True,timeout=120)
(a.output/'incomplete.jsonl').write_bytes(broken.stdout)
assert broken.returncode==2
assert not any(r.get('domain_facts') or r.get('implementation_facts') for r in map(json.loads,broken.stdout.splitlines()))
(a.output/'Fixture.cs').write_text(source)
print('PASS: six ordered request configurations, eight exclusions, incomplete suppression')
