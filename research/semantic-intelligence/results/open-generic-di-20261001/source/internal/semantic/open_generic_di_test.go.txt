package semantic

import (
	"path/filepath"
	"strings"
	"testing"
)

func openDIArtifact(version string) *Artifact {
	a := domainFixture()
	a.Symbols[0].Key.Descriptor = "T:IRepo`1"
	a.Symbols[0].ID = a.Symbols[0].ComputeID()
	api := &a.Symbols[1]
	api.Key.Descriptor = "M:Microsoft.Extensions.DependencyInjection.ServiceCollectionServiceExtensions.AddScoped(Microsoft.Extensions.DependencyInjection.IServiceCollection,System.Type,System.Type)"
	api.ID = api.ComputeID()
	impl := a.Symbols[0]
	impl.Key.Descriptor = "T:Repo`1"
	impl.ID = impl.ComputeID()
	a.Symbols = append(a.Symbols, impl)
	c := &a.Contexts[0]
	c.ExtractorVersion = version
	c.ID = c.ComputeID()
	o := &a.Occurrences[0]
	o.ContextID = c.ID
	o.ID = o.ComputeID()
	b := &a.Bindings[0]
	b.ExtractorVersion = version
	b.OccurrenceID = o.ID
	b.SymbolID = a.Symbols[1].ID
	b.DomainFacts = []DomainFact{{Kind: "di_open_generic_registration_configuration", Rule: "csharp-open-di-v1", EvidenceScope: "compile_time", Lifetime: "scoped", Targets: []DomainTarget{{Role: "service_template", SymbolID: a.Symbols[0].ID}, {Role: "implementation_template", SymbolID: impl.ID}}}}
	b.ID = b.ComputeID()
	return a
}

func TestOpenDIArtifactAndBounds(t *testing.T) {
	for _, version := range []string{"14", "15", "16"} {
		a := openDIArtifact(version)
		if err := a.Validate(); (err == nil) != (version == "15") {
			t.Fatalf("worker %s: %v", version, err)
		}
	}
	a := openDIArtifact("15")
	path := filepath.Join(t.TempDir(), "artifact")
	if err := Write(path, a); err != nil {
		t.Fatal(err)
	}
	if _, err := Read(path, Limits{}); err != nil {
		t.Fatal(err)
	}
	f := a.Bindings[0].DomainFacts[0]
	for _, desc := range []string{"T:Repo", "T:Repo`0", "T:Repo`9", "T:Repo`01", "T:Repo`1.Inner", "T:Repo`1[]", "T:Repo{T}", "T:Outer+Inner`1", "M:Repo`1", "T:`1"} {
		s := a.Symbols[0]
		s.Key.Descriptor = desc
		if ValidateDomainFactTarget(f, "service_template", s) == nil {
			t.Fatalf("invalid template %s", desc)
		}
	}
	if ValidateDomainFactTarget(DomainFact{Kind: "di_registration"}, "service", a.Symbols[0]) == nil {
		t.Fatal("open identity escaped template rule")
	}
	for name, change := range map[string]func(*Artifact){
		"arity": func(a *Artifact) {
			a.Symbols[2].Key.Descriptor = "T:Repo`2"
			a.Symbols[2].ID = a.Symbols[2].ComputeID()
			a.Bindings[0].DomainFacts[0].Targets[1].SymbolID = a.Symbols[2].ID
		},
		"same-template": func(a *Artifact) { a.Bindings[0].DomainFacts[0].Targets[1].SymbolID = a.Symbols[0].ID },
		"role":          func(a *Artifact) { a.Bindings[0].DomainFacts[0].Targets[0].Role = "service" },
		"rule":          func(a *Artifact) { a.Bindings[0].DomainFacts[0].Rule = "csharp-framework-v1" },
		"scope":         func(a *Artifact) { a.Bindings[0].DomainFacts[0].EvidenceScope = "runtime" },
		"lifetime":      func(a *Artifact) { a.Bindings[0].DomainFacts[0].Lifetime = "singleton" },
		"attribute":     func(a *Artifact) { a.Bindings[0].DomainFacts[0].RoutePattern = "/repo" },
		"api-owner": func(a *Artifact) {
			a.Symbols[1].Key.NamespaceKind = "project"
			a.Symbols[1].ID = a.Symbols[1].ComputeID()
			a.Bindings[0].SymbolID = a.Symbols[1].ID
		},
		"api-overload": func(a *Artifact) {
			a.Symbols[1].Key.Descriptor = strings.ReplaceAll(a.Symbols[1].Key.Descriptor, "System.Type,System.Type", "System.Type,System.Func{System.IServiceProvider,System.Object}")
			a.Symbols[1].ID = a.Symbols[1].ComputeID()
			a.Bindings[0].SymbolID = a.Symbols[1].ID
		},
	} {
		t.Run(name, func(t *testing.T) {
			bad := openDIArtifact("15")
			change(bad)
			bad.Bindings[0].ID = bad.Bindings[0].ComputeID()
			if bad.Validate() == nil {
				t.Fatal("invalid open template evidence accepted")
			}
		})
	}
}
