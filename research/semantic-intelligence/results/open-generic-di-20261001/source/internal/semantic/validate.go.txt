package semantic

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"net/url"
	"path"
	"strings"
	"unicode"
)

func digest(s string) bool {
	b, e := hex.DecodeString(s)
	return e == nil && len(b) == 32 && s == strings.ToLower(s)
}
func required(s string) bool { return strings.TrimSpace(s) != "" && !strings.ContainsRune(s, 0) }
func validProjectID(s string) bool {
	if s == "" {
		return true
	}
	if s[0] < '1' || s[0] > '9' {
		return false
	}
	for _, c := range s {
		if c < '0' || c > '9' {
			return false
		}
	}
	return true
}
func validCommit(s string) bool {
	if s == "" {
		return true
	}
	if len(s) != 40 && len(s) != 64 {
		return false
	}
	_, e := hex.DecodeString(s)
	return e == nil
}
func validOrigin(s string) bool {
	if s == "" {
		return true
	}
	if len(s) > 2048 || strings.ContainsAny(s, "\\\r\n\t\x00 ") {
		return false
	}
	for _, r := range s {
		if unicode.IsControl(r) {
			return false
		}
	}
	u, err := url.Parse(s)
	return err == nil && u.Scheme == "https" && u.Hostname() != "" && u.User == nil && u.RawQuery == "" && !u.ForceQuery && u.Fragment == "" && !strings.Contains(s, "#") && u.Path != "" && u.Path != "/"
}
func relative(p string) bool {
	return required(p) && p != "." && !strings.ContainsAny(p, "\\:") && !strings.HasPrefix(p, "/") && path.Clean(p) == p && p != ".." && !strings.HasPrefix(p, "../")
}
func sortedUnique(ss []string) bool {
	for i, s := range ss {
		if !required(s) || (i > 0 && ss[i-1] >= s) {
			return false
		}
	}
	return true
}
func oneOf(s string, values ...string) bool {
	for _, v := range values {
		if s == v {
			return true
		}
	}
	return false
}

// Validate checks identities, evidence shape and referential integrity. It
// cannot establish source-byte authenticity; ingestion must verify raw source
// digests, byte sizes, and UTF-8 span boundaries before constructing an artifact.
func (a *Artifact) Validate() error {
	if a == nil || a.Version != FormatVersion {
		return fmt.Errorf("semantic: unsupported or missing artifact version")
	}
	if len(a.Snapshots) == 0 || len(a.Contexts) == 0 {
		return fmt.Errorf("semantic: snapshots and contexts required")
	}
	seen := map[string]bool{}
	checkID := func(id, want string) error {
		if id != want || seen[id] {
			return fmt.Errorf("semantic: invalid or duplicate ID %q", id)
		}
		seen[id] = true
		return nil
	}
	snapshots := map[string]SourceSnapshot{}
	for _, s := range a.Snapshots {
		if e := checkID(s.ID, s.ComputeID()); e != nil {
			return e
		}
		if !required(s.Repo) || !digest(s.InputFingerprint) || !validProjectID(s.ProjectID) || !validCommit(s.Commit) || !validOrigin(s.Origin) {
			return fmt.Errorf("semantic: invalid snapshot %s", s.ID)
		}
		snapshots[s.ID] = s
	}
	sources := map[string]Source{}
	sourcePaths := map[string]bool{}
	for _, s := range a.Sources {
		if e := checkID(s.ID, s.ComputeID()); e != nil {
			return e
		}
		if _, ok := snapshots[s.SnapshotID]; !ok {
			return fmt.Errorf("semantic: unknown source snapshot")
		}
		key := s.SnapshotID + "\x00" + s.Path
		if !relative(s.Path) || !digest(s.RawSHA256) || sourcePaths[key] {
			return fmt.Errorf("semantic: invalid or duplicate source path %q", s.Path)
		}
		if s.Generated {
			h := sha256.Sum256(s.Content)
			if uint64(len(s.Content)) != s.ByteSize || hex.EncodeToString(h[:]) != s.RawSHA256 {
				return fmt.Errorf("semantic: generated source content mismatch")
			}
		} else if len(s.Content) != 0 {
			return fmt.Errorf("semantic: ordinary source has inline content")
		}
		sourcePaths[key] = true
		sources[s.ID] = s
	}
	contexts := map[string]BuildContext{}
	memberships := map[string]map[string]bool{}
	for _, c := range a.Contexts {
		if c.DependencyBundleSHA256 != "" && !digest(c.DependencyBundleSHA256) {
			return fmt.Errorf("semantic: invalid dependency bundle digest")
		}
		if e := checkID(c.ID, c.ComputeID()); e != nil {
			return e
		}
		if _, ok := snapshots[c.SnapshotID]; !ok {
			return fmt.Errorf("semantic: unknown context snapshot")
		}
		if !relative(c.Project) || !digest(c.InputFingerprint) || !required(c.Extractor) || !required(c.ExtractorVersion) || !oneOf(c.Status, "complete", "incomplete", "unsupported") {
			return fmt.Errorf("semantic: invalid context %s", c.ID)
		}
		if !json.Valid([]byte(c.Capture)) {
			return fmt.Errorf("semantic: invalid context capture")
		}
		h := sha256.Sum256([]byte(c.Capture))
		if hex.EncodeToString(h[:]) != c.InputFingerprint {
			return fmt.Errorf("semantic: context capture digest mismatch")
		}
		if !sortedUnique(c.SourceIDs) || !sortedUnique(c.DependencyContextIDs) {
			return fmt.Errorf("semantic: noncanonical context membership")
		}
		members := map[string]bool{}
		for _, id := range c.SourceIDs {
			s, ok := sources[id]
			if !ok || s.SnapshotID != c.SnapshotID {
				return fmt.Errorf("semantic: invalid context source membership")
			}
			members[id] = true
		}
		if c.Status == "complete" && len(c.Issues) > 0 {
			return fmt.Errorf("semantic: complete context has issues")
		}
		contexts[c.ID] = c
		memberships[c.ID] = members
	}
	visiting, visited := map[string]bool{}, map[string]bool{}
	var visit func(string) error
	visit = func(id string) error {
		if visiting[id] {
			return fmt.Errorf("semantic: dependency cycle")
		}
		if visited[id] {
			return nil
		}
		visiting[id] = true
		c := contexts[id]
		for _, d := range c.DependencyContextIDs {
			dep, ok := contexts[d]
			if !ok {
				return fmt.Errorf("semantic: unknown dependency context")
			}
			if c.Status == "complete" && dep.Status != "complete" {
				return fmt.Errorf("semantic: complete context depends on incomplete context")
			}
			if e := visit(d); e != nil {
				return e
			}
		}
		visiting[id] = false
		visited[id] = true
		return nil
	}
	for id := range contexts {
		if e := visit(id); e != nil {
			return e
		}
	}
	for _, d := range a.Diagnostics {
		if e := checkID(d.ID, d.ComputeID()); e != nil {
			return e
		}
		c, ok := contexts[d.ContextID]
		if !ok || !required(d.Code) || !required(d.Message) || !oneOf(d.Severity, "error", "warning", "info", "hidden") {
			return fmt.Errorf("semantic: invalid diagnostic")
		}
		if c.Status == "complete" && d.Severity == "error" {
			return fmt.Errorf("semantic: complete context has error diagnostic")
		}
		if d.SourceID == "" {
			if d.Offset != 0 || d.Length != 0 {
				return fmt.Errorf("semantic: locationless diagnostic has span")
			}
		} else {
			s, ok := sources[d.SourceID]
			if !ok || !memberships[d.ContextID][d.SourceID] || d.Offset > s.ByteSize || d.Length > s.ByteSize-d.Offset {
				return fmt.Errorf("semantic: invalid diagnostic source span")
			}
		}
	}
	symbols := map[string]bool{}
	symbolRecords := map[string]Symbol{}
	for _, s := range a.Symbols {
		if e := checkID(s.ID, s.ComputeID()); e != nil {
			return e
		}
		k := s.Key
		if !required(k.Language) || k.Language != strings.ToLower(k.Language) || !oneOf(k.NamespaceKind, "project", "assembly", "package") || !required(k.Namespace) || !required(k.Descriptor) || !oneOf(k.DescriptorKind, "documentation_comment_id", "location_fallback", "constructed_interface_method_v1", "constructed_interface_method_v2", "constructed_named_type_v1") {
			return fmt.Errorf("semantic: invalid symbol key")
		}
		if k.DescriptorKind == "constructed_named_type_v1" {
			if e := ValidateConstructedNamedType(k); e != nil {
				return e
			}
		}
		if oneOf(k.DescriptorKind, "constructed_interface_method_v1", "constructed_interface_method_v2") {
			if e := ValidateConstructedInterfaceMethod(k); e != nil {
				return e
			}
		}
		symbols[s.ID] = true
		symbolRecords[s.ID] = s
	}
	occurrences := map[string]Occurrence{}
	for _, o := range a.Occurrences {
		if e := checkID(o.ID, o.ComputeID()); e != nil {
			return e
		}
		s, ok := sources[o.SourceID]
		if !ok || !memberships[o.ContextID][o.SourceID] {
			return fmt.Errorf("semantic: occurrence outside context source membership")
		}
		if !oneOf(o.Role, "declaration", "reference") || !oneOf(o.Kind, "declaration", "constructor_declaration", "name", "invocation", "constructor") || (o.Role == "declaration") != oneOf(o.Kind, "declaration", "constructor_declaration") || o.Length == 0 || o.Offset > s.ByteSize || o.Length > s.ByteSize-o.Offset {
			return fmt.Errorf("semantic: invalid occurrence span or role")
		}
		occurrences[o.ID] = o
	}
	bound := map[string]bool{}
	for _, b := range a.Bindings {
		if e := checkID(b.ID, b.ComputeID()); e != nil {
			return e
		}
		o, ok := occurrences[b.OccurrenceID]
		if !ok || bound[b.OccurrenceID] {
			return fmt.Errorf("semantic: missing or multiply bound occurrence")
		}
		bound[b.OccurrenceID] = true
		c := contexts[o.ContextID]
		if !required(b.Method) || b.Extractor != c.Extractor || b.ExtractorVersion != c.ExtractorVersion || !sortedUnique(b.CandidateSymbolIDs) {
			return fmt.Errorf("semantic: invalid binding evidence")
		}
		if b.EnclosingSymbolID != "" && !symbols[b.EnclosingSymbolID] {
			return fmt.Errorf("semantic: unknown enclosing symbol")
		}
		switch b.Status {
		case "resolved":
			if !symbols[b.SymbolID] || len(b.CandidateSymbolIDs) > 0 {
				return fmt.Errorf("semantic: invalid resolved binding")
			}
		case "ambiguous":
			if b.SymbolID != "" || len(b.CandidateSymbolIDs) < 2 {
				return fmt.Errorf("semantic: invalid ambiguous binding")
			}
		case "unresolved", "unsupported":
			if b.SymbolID != "" {
				return fmt.Errorf("semantic: unresolved binding has symbol")
			}
		default:
			return fmt.Errorf("semantic: invalid binding status")
		}
		for _, id := range b.CandidateSymbolIDs {
			if !symbols[id] {
				return fmt.Errorf("semantic: unknown candidate symbol")
			}
		}
		if e := ValidateDomainFacts(b.DomainFacts); e != nil {
			return e
		}
		if e := ValidateImplementationFacts(b.ImplementationFacts); e != nil {
			return e
		}
		if len(b.ImplementationFacts) > 0 {
			if b.Status != "resolved" || o.Role != "declaration" || o.Kind != "declaration" || c.Status != "complete" || b.Method != "roslyn-semantic-model" || b.Extractor != "msbuild-roslyn" || (b.ExtractorVersion != "6" && b.ExtractorVersion != "7" && b.ExtractorVersion != "8" && b.ExtractorVersion != "9" && (b.ExtractorVersion != "10" && (b.ExtractorVersion != "11" && (b.ExtractorVersion != "12" && (b.ExtractorVersion != "13" && (b.ExtractorVersion != "14" && b.ExtractorVersion != "15")))))) {
				return fmt.Errorf("semantic: implementation facts require complete supported worker declaration")
			}
			for _, f := range b.ImplementationFacts {
				if e := ValidateImplementationExtractor(f, b.ExtractorVersion); e != nil {
					return e
				}
				var selection []Symbol
				for _, id := range f.SymbolIDs()[2:] {
					selection = append(selection, symbolRecords[id])
				}
				if e := ValidateImplementationSymbols(f, symbolRecords[b.SymbolID], symbolRecords[f.InterfaceSymbolID], symbolRecords[f.ImplementingTypeSymbolID], selection...); e != nil {
					return e
				}
			}
		}
		if len(b.DomainFacts) > 0 {
			if b.Status != "resolved" || o.Role != "reference" || c.Status != "complete" || b.Method != "roslyn-semantic-model" || b.Extractor != "msbuild-roslyn" {
				return fmt.Errorf("semantic: domain facts require complete supported compiler binding")
			}
			for _, f := range b.DomainFacts {
				if e := ValidateDomainExtractor(f, b.ExtractorVersion); e != nil {
					return e
				}
				if e := ValidateDomainAPI(f, symbolRecords[b.SymbolID]); e != nil {
					return e
				}
				var targetSymbols []Symbol
				for _, target := range f.Targets {
					s, ok := symbolRecords[target.SymbolID]
					if !ok {
						return fmt.Errorf("semantic: unknown domain target")
					}
					targetSymbols = append(targetSymbols, s)
					if e := ValidateDomainFactTarget(f, target.Role, s); e != nil {
						return e
					}
				}
				if e := ValidateDomainTargetCorrespondence(f, targetSymbols); e != nil {
					return e
				}
			}
		}
	}
	bindingRecords := map[string]Binding{}
	for _, b := range a.Bindings {
		for _, f := range b.ImplementationFacts {
			if f.Forwarding != nil {
				bindingRecords[f.Forwarding.CallOccurrenceID] = Binding{}
			}
		}
	}
	for _, b := range a.Bindings {
		if _, needed := bindingRecords[b.OccurrenceID]; needed {
			bindingRecords[b.OccurrenceID] = b
		}
	}
	for _, b := range a.Bindings {
		for _, f := range b.ImplementationFacts {
			if f.Forwarding != nil {
				own := contexts[occurrences[b.OccurrenceID].ContextID]
				call := occurrences[f.Forwarding.CallOccurrenceID]
				if err := ValidateForwardingWitness(f, own.SnapshotID, call, bindingRecords[call.ID], sources[call.SourceID], contexts[call.ContextID], symbolRecords[f.DefaultTemplateSymbolID], symbolRecords[f.Forwarding.InterfaceSymbolID]); err != nil {
					return err
				}
				seen := map[string]bool{}
				var reaches func(string) bool
				reaches = func(id string) bool {
					if id == call.ContextID {
						return true
					}
					if seen[id] {
						return false
					}
					seen[id] = true
					for _, dep := range contexts[id].DependencyContextIDs {
						if reaches(dep) {
							return true
						}
					}
					return false
				}
				if !reaches(own.ID) {
					return fmt.Errorf("semantic: forwarding witness outside context dependency closure")
				}
			}
		}
	}
	if len(bound) != len(occurrences) {
		return fmt.Errorf("semantic: occurrence missing explicit binding status")
	}
	return nil
}
