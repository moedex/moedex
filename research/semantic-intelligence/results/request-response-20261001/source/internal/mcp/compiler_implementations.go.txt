package mcp

import (
	"bytes"
	"context"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"strings"

	"moedex/internal/semantic"
	"moedex/internal/semanticindex"
)

type compilerImplementationsReader interface {
	Implementations(context.Context, string, []string, int) (semanticindex.ImplementationsResult, error)
}
type compilerImplementationsTool struct{ provider CompilerSessionProvider }

func (*compilerImplementationsTool) Name() string { return "compiler_implementations" }

type CompilerImplementationMatch struct {
	Source                  CompilerBinding `json:"source"`
	ImplementationFactIndex int             `json:"implementation_fact_index"`
}
type CompilerImplementationsResult struct {
	Status                   string                        `json:"status"`
	EvidenceScope            string                        `json:"evidence_scope"`
	SnapshotID               string                        `json:"snapshot_id"`
	ArtifactSHA256           string                        `json:"artifact_sha256"`
	Interface                *CompilerSymbol               `json:"interface,omitempty"`
	ContextIDs               []string                      `json:"context_ids"`
	Contexts                 []CompilerContext             `json:"contexts"`
	Matches                  []CompilerImplementationMatch `json:"matches"`
	Count                    int                           `json:"count"`
	TotalIsExact             bool                          `json:"total_is_exact"`
	Truncated                bool                          `json:"truncated"`
	Limit                    int                           `json:"limit"`
	RowsVisited              int                           `json:"rows_visited"`
	WorkLimit                int                           `json:"work_limit"`
	MaterializationByteLimit int                           `json:"materialization_byte_limit"`
	Error                    *compilerError                `json:"error,omitempty"`
}

func (t *compilerImplementationsTool) Call(ctx context.Context, raw json.RawMessage) (map[string]interface{}, error) {
	out := CompilerImplementationsResult{Status: "semantic_unavailable", EvidenceScope: "compile_time", ContextIDs: []string{}, Contexts: []CompilerContext{}, Matches: []CompilerImplementationMatch{}, Limit: semanticindex.MaxResults, WorkLimit: semanticindex.ContractWorkLimit, MaterializationByteLimit: semanticindex.MaxQueryBytes}
	respond := func(message string, bad bool) map[string]interface{} {
		return StructuredResultWithSnapshot(message, out, bad, SnapshotIdentity{Cacheable: false})
	}
	invalid := func(message string) (map[string]interface{}, error) {
		out.Status = "invalid_arguments"
		out.Error = &compilerError{Code: "invalid_arguments", Message: message}
		return respond(message, true), nil
	}
	var args struct {
		Symbol   string   `json:"symbol_id"`
		Contexts []string `json:"context_ids"`
		Limit    *int     `json:"limit"`
	}
	dec := json.NewDecoder(bytes.NewReader(raw))
	dec.DisallowUnknownFields()
	if dec.Decode(&args) != nil {
		return invalid("Expected symbol_id, optional context_ids, and limit.")
	}
	var extra any
	if dec.Decode(&extra) != io.EOF {
		return invalid("Arguments must be one object.")
	}
	var fields map[string]json.RawMessage
	if json.Unmarshal(raw, &fields) != nil || fields == nil {
		return invalid("Arguments must be an object.")
	}
	digest, err := hex.DecodeString(strings.TrimPrefix(args.Symbol, "symbol:"))
	if !strings.HasPrefix(args.Symbol, "symbol:") || err != nil || len(digest) != 32 || args.Symbol != strings.ToLower(args.Symbol) {
		return invalid("An exact interface method symbol_id is required.")
	}
	if args.Limit != nil {
		out.Limit = *args.Limit
	}
	if _, ok := fields["limit"]; ok && args.Limit == nil {
		return invalid("limit must be 1..100.")
	}
	if out.Limit < 1 || out.Limit > semanticindex.MaxResults {
		return invalid("limit must be 1..100.")
	}
	if _, ok := fields["context_ids"]; ok && (len(args.Contexts) == 0 || len(args.Contexts) > semanticindex.MaxContractContexts) {
		return invalid("context_ids must contain 1..32 unique recorded context IDs.")
	}
	selected := map[string]bool{}
	for _, id := range args.Contexts {
		if !exactContractID(id, "context:") || selected[id] {
			return invalid("context_ids must contain 1..32 unique recorded context IDs.")
		}
		selected[id] = true
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	if t.provider == nil {
		return respond("Compiler evidence is unavailable in this snapshot.", false), nil
	}
	session, err := t.provider.AcquireCompilerSession(ctx)
	if session.Release != nil {
		defer session.Release()
	}
	if err != nil {
		return nil, err
	}
	out.SnapshotID, out.ArtifactSHA256 = session.SnapshotID, session.ArtifactSHA256
	if session.Reader == nil {
		return respond("Compiler evidence is unavailable in this snapshot.", false), nil
	}
	reader, ok := session.Reader.(compilerImplementationsReader)
	if !ok {
		out.Status = "capability_unavailable"
		return respond("This reader lacks interface implementation lookup; binding and definition tools remain available.", false), nil
	}
	q, err := reader.Implementations(ctx, args.Symbol, args.Contexts, out.Limit)
	if err != nil {
		if errors.Is(err, semanticindex.ErrContractContextSelection) {
			out.Status = "invalid_context_selection"
			out.Error = &compilerError{Code: out.Status, Message: err.Error()}
			return respond(err.Error(), true), nil
		}
		return nil, err
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	if !q.Supported {
		out.Status = "index_upgrade_required"
		return respond("Rebuild the semantic index to enable interface-member implementation lookup.", false), nil
	}
	if len(q.Matches) > out.Limit || len(q.Contexts) > semanticindex.MaxContractContexts || len(q.Contexts) > out.Limit || q.RowsVisited < 0 || q.RowsVisited > semanticindex.ContractWorkLimit || q.WorkLimit != semanticindex.ContractWorkLimit {
		return nil, fmt.Errorf("compiler implementations reader exceeded bounds")
	}
	if q.Interface == nil && (len(q.Contexts) != 0 || len(q.Matches) != 0) {
		return nil, fmt.Errorf("compiler implementations returned evidence without interface identity")
	}
	out.Truncated, out.RowsVisited = q.Truncated, q.RowsVisited
	if q.Interface != nil {
		s := q.Interface
		k := s.Key
		name, _, _ := strings.Cut(k.Descriptor, "(")
		if s.ID != args.Symbol || s.ID != s.ComputeID() || k.Language != "csharp" || (k.NamespaceKind != "project" && k.NamespaceKind != "assembly") || strings.TrimSpace(k.Namespace) == "" || strings.ContainsRune(k.Namespace, 0) || !((k.DescriptorKind == "documentation_comment_id" && strings.HasPrefix(name, "M:") && !strings.Contains(name, "`")) || semantic.ValidateConstructedInterfaceMethod(k) == nil) || strings.ContainsRune(k.Descriptor, 0) {
			return nil, fmt.Errorf("compiler implementations returned inconsistent interface identity")
		}
		symbol := compilerSymbol(*s)
		out.Interface = &symbol
	}
	validSource := func(c semanticindex.SourceContext) bool {
		rawHash, hashErr := hex.DecodeString(c.Source.RawSHA256)
		return hashErr == nil && len(rawHash) == 32 && strings.ToLower(c.Source.RawSHA256) == c.Source.RawSHA256 && exactContractID(c.Source.ID, "source:") && exactContractID(c.Snapshot.ID, "snapshot:") && exactContractID(c.Context.ID, "context:") && c.Context.Status == "complete" && c.Context.Extractor == "msbuild-roslyn" && (c.Context.ExtractorVersion == "6" || c.Context.ExtractorVersion == "7" || c.Context.ExtractorVersion == "8" || c.Context.ExtractorVersion == "9" || (c.Context.ExtractorVersion == "10" || c.Context.ExtractorVersion == "11")) && c.Source.SnapshotID == c.Snapshot.ID && c.Context.SnapshotID == c.Snapshot.ID && compilerRelative(c.Snapshot.Repo) && compilerRelative(c.Context.Project) && compilerRelative(c.Source.Path)
	}
	contextIdentity := map[string]string{}
	contexts := map[string]bool{}
	projects := map[string]string{}
	for _, c := range q.Contexts {
		if !validSource(c) || contexts[c.Context.ID] {
			return nil, fmt.Errorf("compiler implementations returned inconsistent context choice")
		}
		contextIdentity[c.Context.ID] = c.Snapshot.ID + "/" + c.Snapshot.Repo + "/" + c.Context.Project

		if len(selected) > 0 && !selected[c.Context.ID] {
			return nil, fmt.Errorf("compiler implementations returned unselected context")
		}
		contexts[c.Context.ID] = true
		key := c.Snapshot.Repo + "/" + c.Context.Project
		if len(selected) > 0 && projects[key] != "" && projects[key] != c.Context.ID {
			return nil, fmt.Errorf("compiler implementations returned incompatible context variants")
		}
		projects[key] = c.Context.ID
		out.Contexts = append(out.Contexts, CompilerContext{Generated: c.Source.Generated, ContextID: c.Context.ID, Project: c.Context.Project, Repo: c.Snapshot.Repo, Path: c.Source.Path, RawSHA256: c.Source.RawSHA256})
	}
	if len(selected) == 0 {
		if len(q.Matches) != 0 {
			return nil, fmt.Errorf("compiler implementations discovery returned undeclared evidence")
		}
		out.Status = "context_required"
		return respond("Select recorded context_ids to inspect implementation declarations. Empty choices do not establish that implementations are absent.", false), nil
	}
	out.ContextIDs = append(out.ContextIDs, args.Contexts...)
	seen := map[string]bool{}
	for _, match := range q.Matches {
		if err := ctx.Err(); err != nil {
			return nil, err
		}
		r := match.Result
		sourceContext := semanticindex.SourceContext{Source: r.Source, Context: r.Context, Snapshot: r.Snapshot}
		if !validSource(sourceContext) || !exactContractID(r.Occurrence.ID, "occurrence:") || contextIdentity[r.Context.ID] != r.Snapshot.ID+"/"+r.Snapshot.Repo+"/"+r.Context.Project {
			return nil, fmt.Errorf("compiler implementations returned inconsistent selected source")
		}

		if out.Interface == nil || !selected[r.Context.ID] || match.ImplementationFactIndex < 0 || match.ImplementationFactIndex >= len(r.Binding.ImplementationFacts) {
			return nil, fmt.Errorf("compiler implementations returned invalid selected evidence")
		}
		binding, err := compilerBinding(r)
		if err != nil {
			return nil, err
		}
		fact := binding.ImplementationFacts[match.ImplementationFactIndex]
		if fact.InterfaceSymbolID != args.Symbol {
			return nil, fmt.Errorf("compiler implementations returned wrong interface target")
		}
		found := false
		for _, s := range binding.ImplementationSymbols {
			if s.ID == args.Symbol {
				found = s == *out.Interface
			}
		}
		if !found {
			return nil, fmt.Errorf("compiler implementations returned mismatched interface descriptor")
		}
		key := r.Snapshot.Repo + "/" + r.Context.Project
		if projects[key] != "" && projects[key] != r.Context.ID {
			return nil, fmt.Errorf("compiler implementations returned incompatible context variants")
		}
		projects[key] = r.Context.ID
		identity := fmt.Sprintf("%s/%s/%d", r.Context.ID, r.Occurrence.ID, match.ImplementationFactIndex)
		if seen[identity] {
			return nil, fmt.Errorf("compiler implementations returned duplicate evidence")
		}
		seen[identity] = true
		out.Matches = append(out.Matches, CompilerImplementationMatch{Source: binding, ImplementationFactIndex: match.ImplementationFactIndex})
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	out.Count = len(out.Matches)
	out.TotalIsExact = !out.Truncated
	out.Status = "ok"
	if out.Count == 0 {
		out.Status = "no_recorded_evidence"
	}
	return respond(fmt.Sprintf("%d captured interface implementation declarations. These compiler correspondences do not select runtime dispatch. Missing evidence is not proof of absence.", out.Count), false), nil
}
func (t *compilerImplementationsTool) Specification() ToolSpecification {
	symbols, binding, _, choices := compilerEvidenceSchemas()
	contexts := arrayOf(stringSchema())
	contexts["minItems"], contexts["maxItems"], contexts["uniqueItems"] = 1, semanticindex.MaxContractContexts, true
	input := objectSchema([]string{"symbol_id"}, map[string]interface{}{"symbol_id": map[string]interface{}{"type": "string", "pattern": "^symbol:[0-9a-f]{64}$"}, "context_ids": contexts, "limit": map[string]interface{}{"type": "integer", "minimum": 1, "maximum": 100}})
	match := objectSchema([]string{"source", "implementation_fact_index"}, map[string]interface{}{"source": binding, "implementation_fact_index": intSchema()})
	props := map[string]interface{}{"status": stringSchema(), "evidence_scope": stringSchema(), "snapshot_id": stringSchema(), "artifact_sha256": stringSchema(), "interface": symbols, "context_ids": arrayOf(stringSchema()), "contexts": arrayOf(choices), "matches": arrayOf(match), "count": intSchema(), "total_is_exact": boolSchema(), "truncated": boolSchema(), "limit": intSchema(), "rows_visited": intSchema(), "work_limit": intSchema(), "materialization_byte_limit": intSchema(), "error": errorPropertySchema()}
	required := []string{"status", "evidence_scope", "snapshot_id", "artifact_sha256", "context_ids", "contexts", "matches", "count", "total_is_exact", "truncated", "limit", "rows_visited", "work_limit", "materialization_byte_limit"}
	return ToolSpecification{Name: t.Name(), Description: "Discover captured source implementations of an exact interface method symbol_id. Omit context_ids for bounded context choices, then explicitly select up to 32 contexts; variants of one repository/project cannot be combined. Each result anchors a method or class declaration with compiler correspondence facts and deduplicated symbols; implementation_fact_index identifies the queried interface fact. Worker7 additionally records closed interface members with qualified named type arguments and explicit default interface implementation templates. Worker8 adds interface_default_selection at concrete class declarations: selected_default_symbol_id identifies the closed default and default_template_symbol_id resolves its source body with compiler_definitions. Worker9 may add a forwarding object for one same-receiver call through an explicit type-parameter cast, including the closed target, class implementation, cast type and exact call witness. Cast success and execution are not established. Default templates alone are not closed class implementations. Open/generic classes, generic methods, inherited class-method mappings and static members remain excluded. Results are compile-time evidence, not runtime dispatch. Limits: 100 results, 10000 posting rows, 2 MiB index materialization (not a serialized response-size ceiling).", InputSchema: input, OutputSchema: objectSchema(required, props)}
}
